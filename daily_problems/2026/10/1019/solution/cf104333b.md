**提示 1：** 每一位是独立的。

**提示 2：** 每一位的组合。

依旧拆成每一位。

我们就是要求所有匹配方式的这一位的和的异或。

我们先预处理 $a_i$ 和 $b_j$ 的组合下这一位是 $1$ 还是 $0$ 。

接下来我们要统计匹配方法，这里 $n\leq 16$ 。

所以考虑 $a$ 的前缀匹配上 $b$ 的 $msk$ 子集的情况下，这一位异或和是 $0/1$ 的方案数，这件事可以 bitmask DP 解决，具体可以看后面的代码。

然后把 $1$ 的方案数乘以 $2^{bit}$ 就是这一位的贡献，加总即可。

时间复杂度为 $\mathcal{O}(\log M\times n\times 2^n)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n = II()
    v1 = LII()
    v2 = LII()
    
    ans = 0
    mod = 10 ** 9 + 7
    
    empty = [[i for i in range(n) if not msk >> i & 1] for msk in range(1 << n)]
    
    for bit in range(30):
        grid = [[0] * n for _ in range(n)]
        
        for x in range(n):
            for y in range(n):
                grid[x][y] = (v1[x] + v2[y]) >> bit & 1
        
        dp = [[0] * (1 << n) for _ in range(2)]
        dp[0][0] = 1
        
        for msk in range(1 << n):
            x = msk.bit_count()
            for y in empty[msk]:
                for i in range(2):
                    ni = i ^ grid[x][y]
                    nmsk = msk | (1 << y)
                    
                    dp[ni][nmsk] += dp[i][msk]
                    dp[ni][nmsk] %= mod
        
        ans += dp[1][(1 << n) - 1] * (1 << bit) % mod
        ans %= mod
    
    print(ans)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n;
	cin >> n;

	vector<int> v1(n), v2(n);

	for (auto &x: v1) cin >> x;
	for (auto &x: v2) cin >> x;

	int ans = 0, mod = 1e9 + 7;

	auto add = [&] (int &x, int y) -> void {
		x += y;
		if (x >= mod) x -= mod;
	};

	for (int bit = 0; bit < 30; bit ++) {
		vector<vector<int>> grid(n, vector<int>(n));

		for (int x = 0; x < n; x ++) {
			for (int y = 0; y < n; y ++) {
				grid[x][y] = (v1[x] + v2[y]) >> bit & 1;
			}
		}

		vector<vector<int>> dp(2, vector<int>(1 << n, 0));
		dp[0][0] = 1 << bit;

		for (int msk = 0; msk < 1 << n; msk ++) {
			int x = __popcount(msk);

			for (int y = 0; y < n; y ++) {
				if (msk >> y & 1) continue;
				for (int i = 0; i < 2; i ++) {
					add(dp[i ^ grid[x][y]][msk | (1 << y)], dp[i][msk]);
				}
			}
		}

		add(ans, dp[1][(1 << n) - 1]);
	}

	cout << ans;

	return 0;
}
```
