**提示 1：** 考虑每个点对答案的贡献。

假设每个点会出现在 $S$ 个点集的凸包上，则答案就是对这个数字求和。

怎么求呢？先把其他店关于极角排序，接下来枚举凸包所在范围的左端射线，设逆时针走 $\pi$ 角度的点数总共是 $v$ （不包含左侧边界），则其中每个点都可以任意选择是否在集合中，因此方案数是 $2^v$ 。求和即可。

最后注意加上 $n$ 个一元集。

时间复杂度为 $\mathcal{O}(n^2\log n)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n = II()
    xs = []
    ys = []
    
    mod = 10 ** 9 + 7
    pw2 = [1] * (n + 1)
    
    for i in range(1, n + 1):
        pw2[i] = pw2[i - 1] * 2 % mod
    
    for _ in range(n):
        x, y = MII()
        xs.append(x)
        ys.append(y)
    
    ans = n
    
    for i in range(n):
        pos = []
        neg = []
        
        for j in range(n):
            if j == i: continue
            
            dx = xs[j] - xs[i]
            dy = ys[j] - ys[i]
            
            if dy > 0 or dy == 0 and dx > 0: pos.append(j)
            else: neg.append(j)
        
        def cmp(x, y):
            dx1, dy1 = xs[x] - xs[i], ys[x] - ys[i]
            dx2, dy2 = xs[y] - xs[i], ys[y] - ys[i]
            return -1 if dx1 * dy2 > dy1 * dx2 else 1
        
        pos.sort(key=cmp_to_key(cmp))
        neg.sort(key=cmp_to_key(cmp))
        
        total = pos + neg
        total *= 2
        
        r = 0
        
        for l in range(n - 1):
            r = fmax(r, l + 1)
            
            while r < l + n - 1:
                dx1, dy1 = xs[total[l]] - xs[i], ys[total[l]] - ys[i]
                dx2, dy2 = xs[total[r]] - xs[i], ys[total[r]] - ys[i]
                
                if dx1 * dy2 > dx2 * dy1: r += 1
                else: break
            
            ans += pw2[r - l - 1]
            ans %= mod
    
    print(ans)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n;
	cin >> n;

	vector<pair<int, int>> pts(n);
	for (auto &[x, y]: pts) cin >> x >> y;

	int mod = 1e9 + 7;

	vector<int> pw2(n + 1);	pw2[0] = 1;
	for (int i = 1; i <= n; i ++) pw2[i] = pw2[i - 1] * 2 % mod;

	int ans = n;

	for (int i = 0; i < n; i ++) {
		vector<int> pos, neg;

		for (int j = 0; j < n; j ++) {
			if (i == j) continue;
			int dx = pts[j].first - pts[i].first;
			int dy = pts[j].second - pts[i].second;
			if (dy > 0 || (dy == 0 && dx > 0)) pos.emplace_back(j);
			else neg.emplace_back(j);
		}

		auto cmp = [&] (int x, int y) -> bool {
			int dx1 = pts[x].first - pts[i].first;
			int dy1 = pts[x].second - pts[i].second;
			int dx2 = pts[y].first - pts[i].first;
			int dy2 = pts[y].second - pts[i].second;
			return 1ll * dx1 * dy2 > 1ll * dx2 * dy1;
		};

		sort(pos.begin(), pos.end(), cmp);
		sort(neg.begin(), neg.end(), cmp);

		vector<int> total;
		total.insert(total.end(), pos.begin(), pos.end());
		total.insert(total.end(), neg.begin(), neg.end());

		for (int i = 0; i < n - 1; i ++) total.emplace_back(total[i]);

		int r = 0;

		for (int l = 0; l < n - 1; l ++) {
			r = max(r, l + 1);

			while (r < l + n - 1) {
				int dx1 = pts[total[l]].first - pts[i].first;
				int dy1 = pts[total[l]].second - pts[i].second;
				int dx2 = pts[total[r]].first - pts[i].first;
				int dy2 = pts[total[r]].second - pts[i].second;

				if (1ll * dx1 * dy2 > 1ll * dx2 * dy1) r ++;
				else break;
			}

			ans += pw2[r - l - 1];
			ans %= mod;
		}
	}

	cout << ans;

	return 0;
}
```
