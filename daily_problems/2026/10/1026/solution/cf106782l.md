**提示 1：** $\lfloor nm/4\rfloor$ ，可以猜猜看怎么查询。

**提示 2：** 证明有点神人，你先猜猜看可以怎么做吧。

首先，我们只能查询 $1/4$ 的位置，所以不如 $2\times 2$ 里面查询一个位置。所以查询 $(2x,2y)$ 的这些位置。

借此我们可以分别得到其周围的某个元素的位置。

对于剩余的部分，如果某个位置的邻接块只有一个，则这个位置的答案也就确定了，类似地不断确定（一个形似拓扑排序的过程，不断将可以断言的块加进来）。这样操作一定能得到答案。

为什么呢？这个证明我并不掌握，我把题解翻译一下吧。

首先，通过查询直接确定的那些骨牌不会形成一个环。

- 如果成环了，则环的长度一定是偶数，则围出来的图形的周长一定是 $4$ 的倍数，根据 Pick 定理，中间部分的面积等于严格内部的点数 + 边界上的点数 / 2 - 1，此时内部没有点，于是面积只可能是奇数——奇数面积不可能由骨牌完全覆盖。

- 因此不可能成环。

而借此，我们可以证明剩余的部分不成环——否则剩余部分形成的环的内部，查询确定的那些部分也就一定成环了，与此前的证明矛盾。

于是，剩余的部分不成环，也就是一系列的森林，森林一定可以用类似拓扑排序的方式形成两两匹配。

但是，尽管你可能想不到这个证明，但是猜到查询这些位置是可以做到的。剩余的部分可能更考验勇气。

时间复杂度为 $\mathcal{O}(nm)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n, m = MII()
    
    ans = [['.'] * m for _ in range(n)]
    
    for i in range(1, n, 2):
        for j in range(1, m, 2):
            print('?', i + 1, j + 1, flush=True)
            ans[i][j] = I()
            
            if ans[i][j] == 'U': ans[i + 1][j] = 'D'
            elif ans[i][j] == 'D': ans[i - 1][j] = 'U'
            elif ans[i][j] == 'L': ans[i][j + 1] = 'R'
            else: ans[i][j - 1] = 'L'
    
    dirs = [(-1, 0), (0, 1), (1, 0), (0, -1)]
    chrs = 'URDL'
    degs = [[0] * m for _ in range(n)]
    
    stk = []
    
    for i in range(n):
        for j in range(m):
            if ans[i][j] != '.': continue
            for di, dj in dirs:
                if 0 <= i + di < n and 0 <= j + dj < m and ans[i + di][j + dj] == '.':
                    degs[i][j] += 1
            
            if degs[i][j] == 1: stk.append((i, j))
    
    while stk:
        x, y = stk.pop()
        
        if ans[x][y] == '.':
            for i in range(4):
                dx, dy = dirs[i]
                if 0 <= x + dx < n and 0 <= y + dy < m and ans[x + dx][y + dy] == '.':
                    ans[x][y] = chrs[i ^ 2]
                    ans[x + dx][y + dy] = chrs[i]
                    
                    degs[x][y] -= 1
                    degs[x + dx][y + dy] -= 1
                    
                    nx, ny = x + dx, y + dy
                    
                    for ndx, ndy in dirs:
                        if 0 <= nx + ndx < n and 0 <= ny + ndy < m and ans[nx + ndx][ny + ndy] == '.':
                            degs[nx + ndx][ny + ndy] -= 1
                            if degs[nx + ndx][ny + ndy] == 1:
                                stk.append((nx + ndx, ny + ndy))
    
    print('!')
    print('\n'.join(''.join(x) for x in ans))
```

C++ 做法如下——

```cpp []
#include <bits/stdc++.h>
using namespace std;

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);

	int n, m;
	cin >> n >> m;

	vector<string> grid(n, string(m, '.'));

	for (int i = 1; i < n; i += 2) {
		for (int j = 1; j < m; j += 2) {
			cout << "? " << i + 1 << ' ' << j + 1 << endl;
			cin >> grid[i][j];

			switch (grid[i][j]) {
				case 'L':
					grid[i][j + 1] = 'R';
					break;
				case 'R':
					grid[i][j - 1] = 'L';
					break;
				case 'U':
					grid[i + 1][j] = 'D';
					break;
				case 'D':
					grid[i - 1][j] = 'U';
					break;
				default:
					break;
			}
		}
	}

	vector<pair<int, int>> dirs = {{-1, 0}, {0, 1}, {1, 0}, {0, -1}};
	string chrs = "URDL";

	vector<vector<int>> degs(n, vector<int>(m, 0));
	vector<pair<int, int>> stk;

	for (int i = 0; i < n; i ++) {
		for (int j = 0; j < m; j ++) {
			if (grid[i][j] != '.') continue;

			for (auto &[di, dj]: dirs) {
				if (i + di >= 0 && i + di < n && j + dj >= 0 && j + dj < m && grid[i + di][j + dj] == '.') {
					degs[i][j] ++;
				}
			}
			if (degs[i][j] == 1) stk.emplace_back(i, j);
		}
	}

	while (!stk.empty()) {
		auto [x, y] = stk.back(); stk.pop_back();

		if (grid[x][y] == '.') {
			for (int i = 0; i < 4; i ++) {
				auto [dx, dy] = dirs[i];

				if (x + dx >= 0 && x + dx < n && y + dy >= 0 && y + dy < m && grid[x + dx][y + dy] == '.') {
					grid[x][y] = chrs[i ^ 2];
					grid[x + dx][y + dy] = chrs[i];

					degs[x][y] --;
					degs[x + dx][y + dy] --;

					int nx = x + dx, ny = y + dy;

					for (auto &[ndx, ndy]: dirs) {
						if (nx + ndx >= 0 && nx + ndx < n && ny + ndy >= 0 && ny + ndy < m && grid[nx + ndx][ny + ndy] == '.') {
							degs[nx + ndx][ny + ndy] --;
							if (degs[nx + ndx][ny + ndy] == 1) {
								stk.emplace_back(nx + ndx, ny + ndy);
							}
						}
					}
				}
			}
		}
	}

	cout << '!' << '\n';
	for (auto &x: grid) cout << x << '\n';

	return 0;
}
```
