**提示 1：** 考虑不得不动的元素，以及其他元素为了使得这些元素能到目标位置需要做什么。

首先，如果一个元素在移动前和移动后既不在同一行，也不在同一列，那一定没办法达成。

否则，首先考虑那些行内需要移动的元素——它们的前后顺序不能变，因为同色的不能在同一个格子内，所以需要一开始就保持正确的顺序。

同列的元素的上下顺序也不能变，也需要一开始就满足从上到下升序。

同时，一个元素从一个位置移动到另一个位置的过程中，途径的那些不动的元素一定只能是另一个颜色的——不然没办法绕开来。这里可以用差分数组来更新（起点到终点的区间）。

那如果横着走或者竖着走都会经过这个格子，那就产生了矛盾了。所以还需要判这件事儿。

而在上述条件都满足的情况下，我们可以让异色的元素先让开位置，再把同色的在这一行排好，再把异色元素挪回来。每行的元素 / 每列的元素都可以这么解决。

时间复杂度为 $\mathcal{O}(n^2)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n = II()
    grid = [LGMI() for _ in range(n)]
    
    tag = [[0] * n for _ in range(n)]
    
    for i in range(n):
        for j in range(n):
            if grid[i][j] // n == i:
                tag[i][j] |= 1
            if grid[i][j] % n == j:
                tag[i][j] |= 2
            
            if tag[i][j] == 0:
                exit(print('No'))
    
    block_tag = [[0] * n for _ in range(n)]
    
    for i in range(n):
        cur = -1
        diff = [0] * n
        
        for j in range(n):
            if tag[i][j] == 1:
                if grid[i][j] < cur:
                    exit(print('No'))
                cur = grid[i][j]
                
                l, r = grid[i][j] % n, j
                if l > r: l, r = r, l
                diff[l + 1] += 1
                diff[r] -= 1
        
        for j in range(1, n):
            diff[j] += diff[j - 1]
        
        for j in range(n):
            if diff[j] and tag[i][j] == 3:
                block_tag[i][j] += 1
    
    for j in range(n):
        cur = -1
        diff = [0] * n
        
        for i in range(n):
            if tag[i][j] == 2:
                if grid[i][j] < cur:
                    exit(print('No'))
                cur = grid[i][j]
                
                l, r = grid[i][j] // n, i
                if l > r: l, r = r, l
                diff[l + 1] += 1
                diff[r] -= 1
        
        for i in range(1, n):
            diff[i] += diff[i - 1]
        
        for i in range(n):
            if diff[i] and tag[i][j] == 3:
                block_tag[i][j] += 1
    
    for i in range(n):
        for j in range(n):
            if block_tag[i][j] == 2:
                exit(print('No'))
    
    print('Yes')
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n;
	cin >> n;

	vector<vector<int>> grid(n, vector<int>(n));
	for (auto &x: grid) for (auto &y: x) cin >> y, y --;

	vector<vector<int>> tag(n, vector<int>(n, 0));

	for (int i = 0; i < n; i ++) {
		for (int j = 0; j < n; j ++) {
			if (grid[i][j] / n == i) tag[i][j] |= 1;
			if (grid[i][j] % n == j) tag[i][j] |= 2;
			if (!tag[i][j]) return cout << "No", 0;
		}
	}

	vector<vector<int>> block_tag(n, vector<int>(n, 0));

	for (int i = 0; i < n; i ++) {
		int cur = -1;
		vector<int> diff(n, 0);

		for (int j = 0; j < n; j ++) {
			if (tag[i][j] == 1) {
				if (grid[i][j] < cur) return cout << "No", 0;
				cur = grid[i][j];

				int l = grid[i][j] % n, r = j;
				if (l > r) swap(l, r);
				diff[l + 1] ++, diff[r] --;
			}
		}

		for (int j = 1; j < n; j ++) diff[j] += diff[j - 1];

		for (int j = 0; j < n; j ++) {
			if (diff[j] && tag[i][j] == 3) {
				block_tag[i][j] ++;
			}
		}
	}

	for (int j = 0; j < n; j ++) {
		int cur = 0;
		vector<int> diff(n, 0);

		for (int i = 0; i < n; i ++) {
			if (tag[i][j] == 2) {
				if (grid[i][j] < cur) return cout << "No", 0;
				cur = grid[i][j];

				int l = grid[i][j] / n, r = i;
				if (l > r) swap(l, r);
				diff[l + 1] ++, diff[r] --;
			}
		}

		for (int i = 1; i < n; i ++) diff[i] += diff[i - 1];

		for (int i = 0; i < n; i ++) {
			if (diff[i] && tag[i][j] == 3) {
				block_tag[i][j] ++;
			}
		}
	}

	for (auto &x: block_tag) for (auto &y: x) if (y == 2) return cout << "No", 0;

	cout << "Yes";

	return 0;
}
```
