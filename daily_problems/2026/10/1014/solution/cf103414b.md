**提示 1：** 找到查询某个位置的数值的两种方案。这样规避一个之后还可以马上用另一个替代方案。

先看提示 1 。

两种方式：一个一个查，查前缀和。

所以第一次操作，我们直接查 $[1,1]$ ，这样，一旦某一次禁止我们查某个前缀，我们就查对应位置的 $[x,x]$ ；一旦禁止我们查 $[x,x]$ ，我们就查 $[1,x]$ 前缀和。

这样我们对于每个下标一定要么查过了 $[x,x]$ ，要么查过了 $[1,x]$ ，这就可以直接恢复每个位置的数值了。

时间复杂度为 $\mathcal{O}(n)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n = II()
    
    res = [0] * (n + 1)
    vis = [0] * (n + 1)
    
    next_query = (-1, -1)
    
    def query(x, y):
        global next_query
        next_query = (-1, -1)
        
        print('?', x, y, flush=True)
        s, vx, vy = MII()
        
        if vis[vy] == 0:
            if vx == vy:
                next_query = (1, vy)
            if vx == 1:
                next_query = (vy, vy)
        
        return s
    
    res[1] = query(1, 1)
    vis[1] = 1
    
    pt = 1
    
    for _ in range(n - 1):
        if next_query != (-1, -1):
            v = next_query[1]
            if next_query[0] == 1:
                vis[v] = 1
            else:
                vis[v] = 2
            res[v] = query(*next_query)
        else:
            while vis[pt]:
                pt += 1
            
            vis[pt] = 2
            res[pt] = query(pt, pt)
    
    ans = [0] * (n + 1)
    ans[1] = res[1]
    
    cur = res[1]
    
    for i in range(2, n + 1):
        if vis[i] == 2: ans[i] = res[i]
        else: ans[i] = res[i] - cur
        cur += ans[i]
    
    print('!')
    print(' '.join(map(str, ans[1:])))
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n;
	cin >> n;

	vector<long long> res(n + 1, 0);
	vector<int> vis(n + 1, 0);

	pair<int, int> next_query = {-1, -1};

	auto query = [&] (int x, int y) -> long long {
		next_query = {-1, -1};
		cout << "? " << x << ' ' << y << endl;

		long long s; int vx, vy;
		cin >> s >> vx >> vy;

		if (!vis[vy]) {
			if (vx == vy) next_query = {1, vy};
			if (vx == 1) next_query = {vy, vy};
		}

		return s;
	};

	res[1] = query(1, 1);
	vis[1] = 1;

	int pt = 1;

	for (int i = 0; i < n - 1; i ++) {
		if (next_query.first != -1) {
			auto [x, y] = next_query;
			if (x == 1) vis[y] = 1;
			else vis[y] = 2;
			res[y] = query(x, y);
		}
		else {
			while (vis[pt]) pt ++;
			vis[pt] = 2;
			res[pt] = query(pt, pt);
		}
	}

	cout << "!\n";
	cout << res[1];
	
	long long cur = res[1];

	for (int i = 2; i <= n; i ++) {
		if (vis[i] == 2) {
			cout << ' ' << res[i];
			cur += res[i];
		}
		else {
			cout << ' ' << res[i] - cur;
			cur = res[i];
		}
	}

	return 0;
}
```
