**提示 1：** 考虑分别统计什么，操作会如何微调统计。

比较特殊的是开头插入元素这件事儿。

- 对行的影响：只有第一行新增了这个新插入的元素，最后一行去掉了最后的元素——其余的是将 $1\sim n-1$ 行的最后一个元素移动到下一行。总共进行的调整次数是 $\mathcal{O}(n)$ 的，可以快速维护。

- 对列的影响：本质上是一些列进行了位置的变化——实际统计数值并未改变。即第 $1\sim n-1$ 列往后移一列，最后一列去掉最后一个元素再在开头加上一个元素，并移动到开头。这里的变动次数也是 $\mathcal{O}(n)$ 的。

这里，整体的列的移动可以通过打标记完成。其余的微调只需找到对应的元素的位置就行。

而修改显然只影响某行某列，是更好维护的。

所以综上，维护每行每列的 $1$ 的个数即可。

时间复杂度为 $\mathcal{O}(n^2+nq)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n, q = MII()
    
    cur = array('i', [0]) * (n * n + q)
    
    cnt_row = [0] * n
    cnt_col = [0] * n
    
    for i in range(n):
        s = [int(c) for c in I()]
        
        for j in range(n):
            cur[i * n + q + j] = s[j]
            cnt_row[i] += s[j]
            cnt_col[(q + j) % n] += s[j]
    
    start_pos = q
    outs = []
    
    for _ in range(q):
        query = LII()
        
        if query[0] == 1:
            row, col = query[1] - 1, query[2] - 1
            target = query[3]
            
            pos = start_pos + row * n + col
            
            cnt_row[row] -= cur[pos]
            cnt_col[pos % n] -= cur[pos]
            
            cur[pos] = target
            
            cnt_row[row] += cur[pos]
            cnt_col[pos % n] += cur[pos]
        
        else:
            for i in range(n):
                cnt_row[i] -= cur[start_pos + i * n + (n - 1)]
            
            for i in range(n - 1):
                cnt_row[i + 1] += cur[start_pos + i * n + (n - 1)]
            
            start_pos -= 1
            cnt_col[start_pos % n] -= cur[start_pos + n * n]
            
            target = query[1]
            
            cur[start_pos] = target
            
            cnt_row[0] += target
            cnt_col[start_pos % n] += target
    
        ans = 0
        
        for x in cnt_row:
            if x == n:
                ans += 1
        
        for x in cnt_col:
            if x == n:
                ans += 1
        
        outs.append(ans)
    
    print('\n'.join(map(str, outs)))
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n, q;
	cin >> n >> q;

	vector<int> cur(n * n + q, 0), cnt_row(n, 0), cnt_col(n, 0);

	for (int i = 0; i < n; i ++) {
		string s;
		cin >> s;

		for (int j = 0; j < n; j ++) {
			cur[i * n + q + j] = s[j] - '0';
			cnt_row[i] += s[j] - '0';
			cnt_col[(q + j) % n] += s[j] - '0';
		}
	}

	int start_pos = q;

	while (q --) {
		int t;
		cin >> t;

		if (t == 1) {
			int row, col, target;
			cin >> row >> col >> target;
			row --, col --;

			int pos = start_pos + row * n + col;

			cnt_row[row] -= cur[pos];
			cnt_col[pos % n] -= cur[pos];

			cur[pos] = target;

			cnt_row[row] += cur[pos];
			cnt_col[pos % n] += cur[pos];
		}
		else {
			for (int i = 0; i < n; i ++) {
				cnt_row[i] -= cur[start_pos + i * n + (n - 1)];
			}

			for (int i = 0; i < n - 1; i ++) {
				cnt_row[i + 1] += cur[start_pos + i * n + (n - 1)];
			}

			start_pos --;
			cnt_col[start_pos % n] -= cur[start_pos + n * n];

			int target;
			cin >> target;

			cur[start_pos] = target;
			cnt_row[0] += target;
			cnt_col[start_pos % n] += target;
		}

		int ans = 0;

		for (auto &x: cnt_row) ans += (x == n);
		for (auto &x: cnt_col) ans += (x == n);

		cout << ans << '\n';
	}

	return 0;
}
```
