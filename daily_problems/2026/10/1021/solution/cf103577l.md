**提示 1：** 分两步。每个元素可能怎么变动？怎么变动最好？

每个树上的元素都可以任意选择 $q$ 个元素的任意子集加进去，所以可选方案都是背包可以凑成的数。

于是先进行可行背包，判断哪些容量是可以凑出来的。

接下来，每个元素都要越小越好，但每个结点的元素会根据其子结点给出限制。

所以从底往根进行树上 DP ，每个结点进行计算数值时，其子结点数值已经确定，我们只需找到一个最小的可以加的数，使得加了之后这个点数值大于等于其子结点数值即可，这件事儿可以二分完成。

时间复杂度为 $\mathcal{O}(n\log(nM)+n^2M/w)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n, q = MII()
    nums = LII()
    
    path = [[] for _ in range(n)]
    for _ in range(n - 1):
        u, v = GMI()
        path[u].append(v)
        path[v].append(u)
    
    vals = LII()
    
    dp = 1
    for x in vals:
        dp |= dp << x
    
    res = [int(x) for x in reversed(bin(dp)[2:])]
    k = len(res)
    
    modi = [i for i in range(k) if res[i]]
    
    parent = [-1] * n
    que = [0]
    
    for u in que:
        for v in path[u]:
            if parent[u] != v:
                parent[v] = u
                que.append(v)
    
    flg = True
    
    for u in reversed(que):
        w = 0
        for v in path[u]:
            if parent[v] == u:
                w = fmax(w, nums[v])
        
        if w <= nums[u]: continue
    
        p = bisect.bisect_left(modi, w - nums[u])
        
        if p == len(modi):
            flg = False
        else:
            nums[u] += modi[p]
    
    print(sum(nums) if flg else -1)
```

C++ 做法如下——

```cpp []
bitset<1000005> bs;

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n, q;
	cin >> n >> q;

	vector<int> nums(n);
	for (auto &x: nums) cin >> x;

	vector<vector<int>> path(n);
	for (int i = 0; i < n - 1; i ++) {
		int u, v;
		cin >> u >> v;
		u --, v --;
		path[u].emplace_back(v);
		path[v].emplace_back(u);
	}

	bs.set(0, 1);

	while (q --) {
		int v;
		cin >> v;
		bs |= bs << v;
	}

	vector<int> possible_values;
	for (int i = 0; i <= 1'000'000; i ++) {
		if (bs[i]) {
			possible_values.emplace_back(i);
		}
	}

	bool flg = true;

	auto dfs = [&] (auto &self, int u, int p) -> void {
		int cur = 0;
		for (auto &v: path[u]) {
			if (v != p) {
				self(self, v, u);
				cur = max(cur, nums[v]);
			}
		}

		int pos = lower_bound(possible_values.begin(), possible_values.end(), cur - nums[u]) - possible_values.begin();

		if (pos == possible_values.size()) flg = false;
		else nums[u] += possible_values[pos];
	};

	dfs(dfs, 0, -1);

	if (!flg) cout << -1;
	else {
		long long ans = 0;
		for (auto &x: nums) ans += x;
		cout << ans;
	}

	return 0;
}
```
