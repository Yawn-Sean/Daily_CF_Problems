**提示 1：** 想想直接做怎么做？然后如何优化？

首先，肯定要求 $1$ 到每个点的最短路的长度。

接下来考虑每个点 $u$ 能以最短路到达的点集，这件事儿可以怎么处理呢？考虑这个点连的边，则对应的新的点 $v$ 如果满足 $dis[v]=dis[u]+1$ 则从 $v$ 可达的点从 $u$ 也可达。

因此，我们可以求所有这样的 $v$ 的并集。这里直接暴力做是 $\mathcal{O}((n+m)k)$ 的。

但这里是很容易的集合操作，所以 bitset 优化即可。

时间复杂度为 $\mathcal{O}((n + m)k/w)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main(): 
    n, m, k = MII()
    nodes = LGMI()
    
    path = [[] for _ in range(n)]
    
    for _ in range(m):
        u, v = GMI()
        path[u].append(v)
        path[v].append(u)
    
    dis = [-1] * n
    dis[0] = 0
    
    que = [0]
    
    for u in que:
        for v in path[u]:
            if dis[v] == -1:
                dis[v] = dis[u] + 1
                que.append(v)
    
    bs = [0] * n
    
    for i in range(k):
        bs[nodes[i]] = 1 << i
    
    for i in reversed(que):
        for j in path[i]:
            if dis[j] == dis[i] + 1:
                bs[i] |= bs[j]
    
    print(max(dis[i] for i in range(n) if bs[i] == bs[0]))
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n, m, k;
	cin >> n >> m >> k;

	vector<int> nodes(k);
	for (auto &x: nodes) cin >> x, x --;

	vector<vector<int>> path(n);
	
	while (m --) {
		int u, v;
		cin >> u >> v;
		u --, v --;
		path[u].emplace_back(v);
		path[v].emplace_back(u);
	}

	vector<int> dis(n, -1);
	dis[0] = 0;

	vector<int> que = {0};
	for (int i = 0; i < que.size(); i ++) {
		int u = que[i];
		for (auto &v: path[u]) {
			if (dis[v] == -1) {
				dis[v] = dis[u] + 1;
				que.emplace_back(v);
			}
		}
	}

	vector<bitset<8000>> bs(n);

	for (int i = 0; i < k; i ++)
		bs[nodes[i]].set(i, 1);

	reverse(que.begin(), que.end());
	for (auto &u: que) {
		for (auto &v: path[u]) {
			if (dis[v] == dis[u] + 1) {
				bs[u] |= bs[v];
			}
		}
	}

	int ans = 0;
	for (int i = 0; i < n; i ++) {
		if (bs[i].count() == k) {
			ans = max(ans, dis[i]);
		}
	}

	cout << ans;

	return 0;
}
```
