**提示 1：** 将三角形刻画为两条边。

**提示 2：** 两个方向的决策是独立的。

把三角形的选取理解为两条直角边，则条件就是选的直角边不能重合。

而这也就说明两个方向是相互独立的。我们只要求每个问号两端至少有一侧是选择的边。同时 # 旁边不能有选择的边。

将每一行 / 列的方案全部相乘就是答案了。

时间复杂度为 $\mathcal{O}(n^2)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n = II()
    grid = [I() for _ in range(n)]
    
    ans = 1
    mod = 998244353
    
    for i in range(n):
        dp0, dp1 = 1, 0
        
        for j in range(n):
            if grid[i][j] == '.':
                dp0, dp1 = (dp0 + dp1) % mod, 0
            elif grid[i][j] == '#':
                if j and grid[i][j - 1] == '#': continue
                dp0, dp1 = 0, dp0
            else:
                dp0, dp1 = dp0, (dp0 + dp1) % mod
        
        ans = ans * (dp0 + dp1) % mod
    
    for j in range(n):
        dp0, dp1 = 1, 0
        
        for i in range(n):
            if grid[i][j] == '.':
                dp0, dp1 = (dp0 + dp1) % mod, 0
            elif grid[i][j] == '#':
                if i and grid[i - 1][j] == '#': continue
                dp0, dp1 = 0, dp0
            else:
                dp0, dp1 = dp0, (dp0 + dp1) % mod
        
        ans = ans * (dp0 + dp1) % mod
    
    print(ans)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n;
	cin >> n;

	vector<string> grid(n);
	for (auto &s: grid) cin >> s;

	int ans = 1, mod = 998244353;

	for (int i = 0; i < n; i ++) {
		int dp0 = 1, dp1 = 0;

		for (int j = 0; j < n; j ++) {
			if (grid[i][j] == '.') {
				dp0 = (dp0 + dp1) % mod;
				dp1 = 0;
			}
			else if (grid[i][j] == '#') {
				if (j && grid[i][j - 1] == '#') continue;
				dp1 = dp0;
				dp0 = 0;
			}
			else {
				dp1 = (dp0 + dp1) % mod;
			}
		}

		ans = 1ll * ans * (dp0 + dp1) % mod;
	}

	for (int j = 0; j < n; j ++) {
		int dp0 = 1, dp1 = 0;

		for (int i = 0; i < n; i ++) {
			if (grid[i][j] == '.') {
				dp0 = (dp0 + dp1) % mod;
				dp1 = 0;
			}
			else if (grid[i][j] == '#') {
				if (i && grid[i - 1][j] == '#') continue;
				dp1 = dp0;
				dp0 = 0;
			}
			else {
				dp1 = (dp0 + dp1) % mod;
			}
		}

		ans = 1ll * ans * (dp0 + dp1) % mod;
	}

	cout << ans;

	return 0;
}
```
