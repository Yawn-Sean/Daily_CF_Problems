**提示 1：** 实则区间 DP 。如何进行呢？

因为每次走到一个点我们都可以知道在左边还是右边，所以备选的点集是一个区间。

所以用区间 DP 考虑的话，就是 $dp[l][r]$ 的最小成本。刚刚确定在这个区间的时候，你一定位于区间的某个端点，所以需要分在左端点还是右端点的两种情况。

接下来考虑下一次走到区间内部会走到哪，则根据这个位置，可以将之后的备选分为左右两侧的两个区间。因为这里距离并不单调，所以你只可能选择枚举中间的分割点。

这样，我们就得到了类似 $dp[l][i], dp[i][r]$ 推 $dp[l][r]$ 的结构，总共状态数 $\mathcal{O}(n^2)$ ，转移的 $i$ 有 $\mathcal{O}(n)$ 个，因此总复杂度为 $\mathcal{O}(n^3)$ ，这样并没有办法通过。

怎么办呢？实际上，对于同一个区间长度， DP 的结果是完全一致的：因为你总能以一样的成本走到对应的分割点。这主要是因为最大步长不超过 $n/2$ ，于是对于一个步长集，你总能找到一个排列，使得你到达的位置的范围始终在 $1\sim n$ 之间（偏大了就往小走之类的贪心就行）。

所以你实际上的状态数是 $\mathcal{O}(n)$ 的，这样复杂度就够了。

最后，你走到 $+\Delta$ 位置的最小成本，你需要在最开始的时候预处理，用不带堆优化的 Dijkstra 在完全图中跑就可以达到 $\mathcal{O}(n^2)$ 复杂度。

时间复杂度为 $\mathcal{O}(n^2)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n, J = MII()
    upward = LII()
    downward = LII()
    
    dis = [inf] * (2 * n + 1)
    vis = [0] * (2 * n + 1)
    
    dis[n] = 0
    vis[n] = 1
    
    chosen = n
    
    for _ in range(2 * n):
        for i in range(J):
            u = chosen + i + 1
            if u <= 2 * n and not vis[u]:
                dis[u] = fmin(dis[u], dis[chosen] + upward[i])
            
            u = chosen - i - 1
            if u >= 0 and not vis[u]:
                dis[u] = fmin(dis[u], dis[chosen] + downward[i])
        
        nchosen = -1
        for u in range(2 * n + 1):
            if not vis[u] and (nchosen == -1 or dis[u] < dis[nchosen]):
                nchosen = u
        
        chosen = nchosen
        vis[chosen] = 1
    
    dp_l = [inf] * (n + 1)
    dp_r = [inf] * (n + 1)
    
    dp_l[1] = 0
    dp_r[1] = 0
    
    for i in range(2, n + 1):
        for j in range(1, i):
            v = fmax(dp_r[j], dp_l[i - j])
            dp_l[i] = fmin(dp_l[i], v + dis[n + j])
            dp_r[i] = fmin(dp_r[i], v + dis[n - (i - j)])
    
    print(dp_l[n])
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n, J;
	cin >> n >> J;

	vector<int> upward(J + 1, 0), downward(J + 1, 0);

	for (int i = 1; i <= J; i ++) cin >> upward[i];
	for (int i = 1; i <= J; i ++) cin >> downward[i];

	vector<int> dis(2 * n + 1, (int)1e9), vis(2 * n + 1, 0);

	dis[n] = 0, vis[n] = 1;
	int chosen = n;

	for (int x = 0; x < 2 * n; x ++) {
		for (int i = 1; i <= J; i ++) {
			int u;
			u = chosen + i;
			if (u <= 2 * n) dis[u] = min(dis[u], dis[chosen] + upward[i]);
			u = chosen - i;
			if (u >= 0) dis[u] = min(dis[u], dis[chosen] + downward[i]);
		}

		int nchosen = -1;
		for (int u = 0; u <= 2 * n; u ++) {
			if (!vis[u] && (nchosen == -1 || dis[u] < dis[nchosen])) {
				nchosen = u;
			}
		}

		chosen = nchosen;
		vis[chosen] = 1;
	}

	vector<int> dp_l(n + 1, (int)1e9), dp_r(n + 1, (int)1e9);
	dp_l[1] = 0, dp_r[1] = 0;

	for (int i = 2; i <= n; i ++) {
		for (int j = 1; j < i; j ++) {
			int v = max(dp_r[j], dp_l[i - j]);
			dp_l[i] = min(dp_l[i], v + dis[n + j]);
			dp_r[i] = min(dp_r[i], v + dis[n - (i - j)]);
		}
	}

	cout << dp_l[n];

	return 0;
}
```
