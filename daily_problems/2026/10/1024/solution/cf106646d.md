**提示 1：** 基准是什么？在这个基础上要最优化什么？

直接把所有图形的宽度相加，就是基础答案。

而剩余的部分就是省下来的，来自相邻的多边形横坐标重合的部分。（只需考虑相邻的，因为高度一致，不可能一个横坐标被三个图形覆盖）

所以我们先预处理，找到 $i$ 在左 $j$ 在右的情况下能省下来的横坐标的长度，接下来再用 bitmask DP 得到最后一个多边形为 $i$ ，当前选择的多边形为 $msk$ 的情况下的最大节省长度，即可得到总的节省长度。

后面的 DP 是很典型的。前面的部分，因为总是一个点在一条边上，所以枚举一个点（可能是左边的多边形也可能是右边的多边形），找到另一个多边形的对应高度的边，找其横坐标之差即可。这个坐标差的最值是我们唯一需要关注的。

因为从低到高枚举点时，边可以用一个指针维护，所以这里可以是线性的。但这里题目数据范围较小，你可以不优化。

时间复杂度为 $\mathcal{O}(n^2k+n^22^n)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n, h = MII()
    
    ans = 0
    
    lefts = [[] for _ in range(n)]
    rights = [[] for _ in range(n)]
    
    for i in range(n):
        k = II()
        polygon = [tuple(MII()) for _ in range(k)]
        
        pt = min(polygon, key=lambda x: (x[1], x[0]))
        idx = polygon.index(pt)
        pts = []
        
        while True:
            pts.append(polygon[idx])
            if polygon[idx][1] == h: break
            idx = (idx - 1) % k
        
        lefts[i] = pts
        
        pt = max(polygon, key=lambda x: (x[1], x[0]))
        idx = polygon.index(pt)
        pts = []
        
        while True:
            pts.append(polygon[idx])
            if polygon[idx][1] == 0: break
            idx = (idx - 1) % k
        
        pts.reverse()
        rights[i] = pts
        
        ans += max(polygon)[0] - min(polygon)[0]
    
    grid = [[0] * n for _ in range(n)]
    
    for i in range(n):
        for j in range(n):
            if i != j:
                left = rights[i]
                right = lefts[j]
                
                base = min(right)[0] - max(left)[0]
                real = inf
                
                pt = 0
                
                for idx in range(len(left)):
                    while pt + 1 < len(right) and right[pt + 1][1] < left[idx][1]:
                        pt += 1
                    
                    x, y = left[idx]
                    x1, y1 = right[pt]
                    x2, y2 = right[pt + 1]
                    
                    cur_x = (x1 * (y2 - y1) + (x2 - x1) * (y - y1)) / (y2 - y1)
                    real = fmin(real, cur_x - x)
                
                pt = 0
                
                for idx in range(len(right)):
                    while pt + 1 < len(left) and left[pt + 1][1] < right[idx][1]:
                        pt += 1
                    
                    x, y = right[idx]
                    x1, y1 = left[pt]
                    x2, y2 = left[pt + 1]
                    
                    cur_x = (x1 * (y2 - y1) + (x2 - x1) * (y - y1)) / (y2 - y1)
                    real = fmin(real, x - cur_x)
                
                grid[i][j] = real - base
    
    dp = [[0] * (1 << n) for _ in range(n)]
    
    for i in range(1 << n):
        for j in range(n):
            if i >> j & 1:
                for k in range(n):
                    if i >> k & 1: continue
                    ni = i | (1 << k)
                    dp[k][ni] = fmax(dp[k][ni], dp[j][i] + grid[j][k])
    
    print(ans - max(x[-1] for x in dp))
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n, h;
	cin >> n >> h;

	int ans = 0;

	vector<vector<pair<int, int>>> lefts(n), rights(n);

	for (int i = 0; i < n; i ++) {
		int k;
		cin >> k;

		vector<pair<int, int>> polygon(k);
		int mx = -1e9, mn = 1e9;

		for (auto &[x, y]: polygon) cin >> x >> y, mx = max(mx, x), mn = min(mn, x);
		ans += mx - mn;

		{
			int idx = min_element(polygon.begin(), polygon.end(), [&] (pair<int, int> &x, pair<int, int> &y) {
				return make_pair(x.second, x.first) < make_pair(y.second, y.first);
			}) - polygon.begin();

			vector<pair<int, int>> pts;

			while (true) {
				pts.emplace_back(polygon[idx]);
				if (polygon[idx].second == h) break;
				idx = (idx + k - 1) % k;
			}

			lefts[i] = pts;
		}

		{
			int idx = max_element(polygon.begin(), polygon.end(), [&] (pair<int, int> &x, pair<int, int> &y) {
				return make_pair(x.second, x.first) < make_pair(y.second, y.first);
			}) - polygon.begin();

			vector<pair<int, int>> pts;

			while (true) {
				pts.emplace_back(polygon[idx]);
				if (polygon[idx].second == 0) break;
				idx = (idx + k - 1) % k;
			}
			reverse(pts.begin(), pts.end());

			rights[i] = pts;
		}
	}

	vector<vector<long double>> grid(n, vector<long double>(n, 0));

	for (int i = 0; i < n; i ++) {
		for (int j = 0; j < n; j ++) {
			if (i != j) {
				auto left = rights[i];
				auto right = lefts[j];

				int base = min_element(right.begin(), right.end())->first - max_element(left.begin(), left.end())->first;

				long double real = 1e9;

				{
					int pt = 0;

					for (int idx = 0; idx < left.size(); idx ++) {
						while (pt + 1 < right.size() && right[pt + 1].second < left[idx].second) pt ++;

						auto [x, y] = left[idx];
						auto [x1, y1] = right[pt];
						auto [x2, y2] = right[pt + 1];

						long double cur_x = (long double)(x1 * (y2 - y1) + (x2 - x1) * (y - y1)) / (y2 - y1);
						real = min(real, cur_x - x);
					}
				}

				{
					int pt = 0;

					for (int idx = 0; idx < right.size(); idx ++) {
						while (pt + 1 < left.size() && left[pt + 1].second < right[idx].second) pt ++;

						auto [x, y] = right[idx];
						auto [x1, y1] = left[pt];
						auto [x2, y2] = left[pt + 1];

						long double cur_x = (long double)(x1 * (y2 - y1) + (x2 - x1) * (y - y1)) / (y2 - y1);
						real = min(real, x - cur_x);
					}
				}

				grid[i][j] = real - base;
			}
		}
	}

	vector<vector<long double>> dp(n, vector<long double>(1 << n, 0));

	for (int i = 0; i < (1 << n); i ++) {
		for (int j = 0; j < n; j ++) {
			if (i >> j & 1) {
				for (int k = 0; k < n; k ++) {
					if (i >> k & 1) continue;
					dp[k][i | (1 << k)] = max(dp[k][i | (1 << k)], dp[j][i] + grid[j][k]);
				}
			}
		}
	}

	long double max_saved = 0;

	for (auto &x: dp) max_saved = max(max_saved, x.back());

	cout << fixed << setprecision(15) << ans - max_saved;

	return 0;
}
```
