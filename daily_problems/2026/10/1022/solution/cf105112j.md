**提示 1：** 考虑一个遍历顺序，其对应的角度应该怎么选？

**提示 2：** 枚举可能的角度。

首先，考虑某一个遍历顺序下的最优坐标系角度。不妨设其中的一段是水平的，则发现其最小值是 $d\times (|\sin\theta|+|\cos\theta|)$ 的类似的形式。这类函数平移后相加，相当于一系列三角函数的绝对值相加，最值时，一定满足其中某个三角函数等于 $0$ 。

于是最优的角度一定跟其中的某一段贡献。

因此考虑枚举 $\mathcal{O}(n^2)$ 个方向，接下来使用 bitmask DP 做旅行商问题即可。

然后你计算的时候可以不用浮点数，对于方向 $(dx,dy)$ 你可以计算 $(dx,dy),(-dy,dx)$ 两个方向上的投影，这样你先算点积的绝对值的和的最小值，最后再除以 $\sqrt{dx^2+dy^2}$ 即可。这样可以保证精度损失尽可能小。

时间复杂度为 $\mathcal{O}(n^42^n)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n = II()
    xs = []
    ys = []
    
    for _ in range(n):
        x, y = MII()
        xs.append(x)
        ys.append(y)
    
    ans = inf
    
    for i in range(n):
        for j in range(i):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            
            vx = -dy
            vy = dx
            
            grid = [[0] * n for _ in range(n)]
            
            for x in range(n):
                for y in range(n):
                    grid[x][y] += abs((xs[x] - xs[y]) * dx + (ys[x] - ys[y]) * dy)
                    grid[x][y] += abs((xs[x] - xs[y]) * vx + (ys[x] - ys[y]) * vy)
            
            dp = [[inf] * (1 << n) for _ in range(n)]
            
            for x in range(n):
                dp[x][1 << x] = 0
            
            for msk in range(1 << n):
                for x in range(n):
                    if dp[x][msk] < inf:
                        for y in range(n):
                            if msk >> y & 1: continue
                            nmsk = msk | (1 << y)
                            dp[y][nmsk] = fmin(dp[y][nmsk], dp[x][msk] + grid[x][y])
            
            res = inf
            for x in range(n):
                res = fmin(res, dp[x][-1])
            
            ans = fmin(ans, res / math.hypot(dx, dy))
    
    print(ans)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n;
	cin >> n;

	vector<pair<int, int>> pts(n);
	for (auto &[x, y]: pts) cin >> x >> y;

	long double total_ans = 1e9;

	for (int i = 0; i < n; i ++) {
		for (int j = 0; j < i; j ++) {
			auto [x1, y1] = pts[i];
			auto [x2, y2] = pts[j];

			int dx1 = x2 - x1, dy1 = y2 - y1;
			int dx2 = -dy1, dy2 = dx1;

			vector<vector<long long>> grid(n, vector<long long>(n, 0));

			for (int x = 0; x < n; x ++ ){
				for (int y = 0; y < n; y ++) {
					auto [v1, w1] = pts[x];
					auto [v2, w2] = pts[y];
					grid[x][y] += abs(1ll * dx1 * (v1 - v2) + 1ll * dy1 * (w1 - w2));
					grid[x][y] += abs(1ll * dx2 * (v1 - v2) + 1ll * dy2 * (w1 - w2));
				}
			}

			vector<vector<long long>> dp(n, vector<long long>(1 << n, (long long)1e18));

			for (int x = 0; x < n; x ++) dp[x][1 << x] = 0;

			for (int msk = 0; msk < (1 << n); msk ++) {
				for (int x = 0; x < n; x ++) {
					if (msk >> x & 1) {
						for (int y = 0; y < n; y ++) {
							if (msk >> y & 1) continue;
							dp[y][msk | (1 << y)] = min(dp[y][msk | (1 << y)], dp[x][msk] + grid[x][y]);
						}
					}
				}
			}

			long long ans = 1e18;

			for (auto &v: dp) ans = min(ans, v.back());

			total_ans = min(total_ans, ans / hypotl(dx1, dy1));
		}
	}

	cout << fixed << setprecision(15) << total_ans;

	return 0;
}
```
