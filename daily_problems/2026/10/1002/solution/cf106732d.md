**提示 1：** DP ，如何找到转移区间？

右端点固定时，左端点越靠左，MEX 数值越大（单调不减）。

所以，考虑到第 $i$ 个位置截断的方案数是 $dp[i]$ ，则考虑最后一段的左端点，一定形成一个连续的区间，因此我们只需将这个区间内的总方案数转移到 $dp[i]$ 即可。

如何找到这个区间呢？我们可以维护 MEX 不超过 $x$ 的最小左端点，可以用一个数组，统计严格小于 $x$ 的元素出现了多少个，一旦有元素没有出现，这个位置就开始合法了。然后用 $x=r$ 和 $x=l-1$ 的两个端点即可确定。

时间复杂度为 $\mathcal{O}(n\log n)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n, l, r = MII()
    nums = LII()
    r += 1
    
    mod = 998244353
    
    v1 = [0] * l
    v2 = [0] * r
    
    p1 = 0
    p2 = 0
    
    c1 = 0
    c2 = 0
    
    fen = FenwickTree(n + 1)
    fen.add(0, 1)
    
    for i in range(n):
        if nums[i] < l:
            if v1[nums[i]] == 0: c1 += 1
            v1[nums[i]] += 1
        
        if nums[i] < r:
            if v2[nums[i]] == 0: c2 += 1
            v2[nums[i]] += 1
        
        while p1 <= i and c1 == l:
            if nums[p1] < l:
                v1[nums[p1]] -= 1
                if v1[nums[p1]] == 0:
                    c1 -= 1
            p1 += 1
        
        while p2 <= i and c2 == r:
            if nums[p2] < r:
                v2[nums[p2]] -= 1
                if v2[nums[p2]] == 0:
                    c2 -= 1
            p2 += 1
        
        fen.add(i + 1, fen.rsum(p2, p1 - 1) % mod)
    
    print(fen.rsum(n, n))
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n, l, r, mod = 998244353;
	cin >> n >> l >> r;

	vector<int> nums(n);
	for (auto &x: nums) cin >> x;

	r ++;

	vector<int> v1(l, 0), v2(r, 0);
	int p1 = 0, p2 = 0;
	int c1 = 0, c2 = 0;

	atcoder::fenwick_tree<long long> fen(n + 1);
	fen.add(0, 1);

	for (int i = 0; i < n; i ++) {
		if (nums[i] < l) {
			if (!v1[nums[i]]) c1 ++;
			v1[nums[i]] ++;
		}

		if (nums[i] < r) {
			if (!v2[nums[i]]) c2 ++;
			v2[nums[i]] ++;
		}

		while (p1 <= i && c1 == l) {
			if (nums[p1] < l) {
				v1[nums[p1]] --;
				if (!v1[nums[p1]]) c1 --;
			}
			p1 ++;
		}

		while (p2 <= i && c2 == r) {
			if (nums[p2] < r) {
				v2[nums[p2]] --;
				if (!v2[nums[p2]]) c2 --;
			}
			p2 ++;
		}

		fen.add(i + 1, fen.sum(p2, p1) % mod);
	}

	cout << fen.sum(n, n + 1);

	return 0;
}
```
