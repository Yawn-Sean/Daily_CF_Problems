**提示 1：** 找清楚条件。

这里东南西北需要结合给出的样例确认，实则是上北下南左西右东。

两个条件：

- 到目标点的最短路长度是字符串的长度。

- 到目标点的最短路可以每一项都跟给定的方向不同。

所以我们先求到每个点的最短路。

接下来，假设我们求得了所有走 $k$ 步满足条件的位置，即最短路长度是 $k$ ，且前 $k$ 步可以完全不等于给定操作的可达位置。

接下来 $k+1$ 步的可行位置一定从这些位置出发。我们从这些位置出发，枚举三个非给定操作的方向，看对应的位置是否最短路长度为 $k+1$ ，如果是，将对应位置打上标记即可。

最后将所有步长为 $|s|$ 的标记位置记为答案即可。

时间复杂度为 $\mathcal{O}(nm+|s|)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    m, n = MII()
    tmp = [I() for _ in range(n)]
    ops = I()
    
    grid = [[0] * m for _ in range(n)]
    sx, sy = -1, -1
    
    for i in range(n):
        for j in range(m):
            if tmp[i][j] == '#': grid[i][j] = 1
            elif tmp[i][j] == 'S': sx, sy = i, j
    
    dis = [[-1] * m for _ in range(n)]
    
    def f(x, y): return x * m + y
    
    dis[sx][sy] = 0
    que = [f(sx, sy)]
    dirs = [(-1, 0), (0, 1), (1, 0), (0, -1)]
    
    for u in que:
        x, y = divmod(u, m)
        
        for dx, dy in dirs:
            if grid[x + dx][y + dy]: continue
            if dis[x + dx][y + dy] == -1:
                dis[x + dx][y + dy] = dis[x][y] + 1
                que.append(f(x + dx, y + dy))
    
    d = {
        'N': (-1, 0),
        'S': (1, 0),
        'W': (0, -1),
        'E': (0, 1)
    }
    
    l = len(ops)
    
    ans = [[0] * m for _ in range(n)]
    ans[sx][sy] = 1
    
    for u in que:
        x, y = divmod(u, m)
        
        if ans[x][y] and dis[x][y] < l:
            for c, (dx, dy) in d.items():
                if c != ops[dis[x][y]] and dis[x + dx][y + dy] == dis[x][y] + 1:
                    ans[x + dx][y + dy] = 1
    
    print('\n'.join(''.join(('!' if dis[i][j] == l and ans[i][j] else '.') if tmp[i][j] == '.' else tmp[i][j] for j in range(m)) for i in range(n)))
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n, m;
	cin >> m >> n;

	vector<string> grid(n);
	for (auto &v: grid) cin >> v;

	string ops;
	cin >> ops;

	int l = ops.size();

	int sx = -1, sy = -1;

	for (int i = 0; i < n; i ++) {
		for (int j = 0; j < m; j ++) {
			if (grid[i][j] == 'S') {
				sx = i, sy = j;
			}
		}
	}

	vector<pair<int, int>> que;
	vector<pair<int, int>> dirs = {{-1, 0}, {0, 1}, {1, 0}, {0, -1}};
	vector<vector<int>> dis(n, vector<int>(m, -1));

	que.emplace_back(sx, sy);
	dis[sx][sy] = 0;

	for (int i = 0; i < que.size(); i ++) {
		auto [x, y] = que[i];

		for (auto &[dx, dy]: dirs) {
			if (grid[x + dx][y + dy] == '#') continue;
			if (dis[x + dx][y + dy] == -1) {
				dis[x + dx][y + dy] = dis[x][y] + 1;
				que.emplace_back(x + dx, y + dy);
			}
		}
	}

	map<char, pair<int, int>> mp;
	mp['N'] = {-1, 0};
	mp['S'] = {1, 0};
	mp['W'] = {0, -1};
	mp['E'] = {0, 1};

	vector<vector<int>> ans(n, vector<int>(m, 0));
	ans[sx][sy] = 1;

	for (auto &[x, y]: que) {
		if (ans[x][y]) {
			if (dis[x][y] < l) {
				for (auto &[c, dir]: mp) {
					if (c != ops[dis[x][y]]) {
						auto [dx, dy] = dir;
						if (dis[x + dx][y + dy] == dis[x][y] + 1) {
							ans[x + dx][y + dy] = 1;
						}
					}
				}
			}
			else grid[x][y] = '!';
		}
	}

	for (auto &v: grid) cout << v << '\n';

	return 0;
}
```
