**提示 1：** 偶数更特殊，考虑偶数咋选。

如果你不选偶数，显然可以。你挑选奇数数码最多的几个就好。

如果你选了 $\bmod 4$ 余 $2$ 的数，那么别的类别的（奇数或者 $4$ 的倍数）都选不了了，因为 $奇数+2$ 结尾 和 $2+4的倍数$ 结尾都一定是 $4$ 的倍数。所以看看这个的数量。

如果你选了 $4$ 的倍数，你其实也最多只能选一个，除此之外只能选奇数。所以不妨和只选奇数的放到一起考虑。我们先考虑奇数数量最多的 $k$ 个数。此时你最多把其中一个数替换为奇数数量一样多的且多一个 $4$ 的倍数的元素，因此枚举剩余的那些数看看要不要替换。

替换结束后，再筛出所有的奇数。如果没有奇数，其实你也放不进 $4$ 的倍数，此时输出 empty；否则将所有的奇数搭配上某一个 $4$ 的倍数即可。

时间复杂度为 $\mathcal{O}(n+M)$ ，如果你用基数排序之类的。

### 具体代码如下——

Python 做法如下——

```Python []
def main(): 
    n, k = MII()
    cards = [[int(c) for c in I()] for _ in range(n)]
    
    ans = []
    chosen = []
    
    cnts = []
    for card in cards:
        c = 0
        for x in card:
            if x % 4 == 2:
                c += 1
        cnts.append(c)
    
    st_range = sorted(range(n), key=lambda x: -cnts[x])
    
    tmp = []
    idxs = []
    for i in st_range[:k]:
        idxs.append(i)
        for x in cards[i]:
            if x % 4 == 2:
                tmp.append(x)
    
    if len(tmp) > len(ans):
        ans = tmp
        chosen = idxs
    
    cnts = []
    for card in cards:
        c = 0
        for x in card:
            if x % 2 == 1:
                c += 1
        cnts.append(c)
    
    st_range = sorted(range(n), key=lambda x: -cnts[x])
    for i in range(k, n):
        idx = st_range[i]
        if cnts[idx] == cnts[st_range[k - 1]]:
            flg = False
            for x in cards[idx]:
                if x % 4 == 0:
                    st_range[k - 1], st_range[i] = st_range[i], st_range[k - 1]
                    flg = True
                    break
            
            if flg: break
    
    tmp = []
    idxs = []
    w = -1
    
    for i in st_range[:k]:
        idxs.append(i)
        for x in cards[i]:
            if x % 2 == 1:
                tmp.append(x)
            elif x % 4 == 0:
                w = x
    
    if w != -1 and tmp:
        tmp.append(w)
    
    if len(tmp) > len(ans):
        ans = tmp
        chosen = idxs
    
    if ans:
        print(' '.join(str(x + 1) for x in chosen))
        print(''.join(map(str, ans)))
    else:
        print(' '.join(map(str, range(1, k + 1))))
        print('empty')
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int n, k;
	cin >> n >> k;

	vector<string> strs(n);
	for (auto &x: strs) cin >> x;

	string ans;
	vector<int> chosen, order(n);
	iota(order.begin(), order.end(), 0);

	{
		vector<int> cnt;
		for (auto &s: strs) {
			int c = 0;
			for (auto &x: s) {
				if ((x - '0') % 4 == 2) {
					c ++;
				}
			}
			cnt.emplace_back(c);
		}

		sort(order.begin(), order.end(), [&] (int i, int j) {return cnt[i] > cnt[j];});

		string tmp;
		vector<int> idxs;

		for (int i = 0; i < k; i ++) {
			idxs.emplace_back(order[i]);
			for (auto &x: strs[order[i]]) {
				if ((x - '0') % 4 == 2) {
					tmp += x;
				}
			}
		}

		if (tmp.size() > ans.size()) {
			ans = tmp;
			chosen = idxs;
		}
	}

	{
		vector<int> cnt;
		for (auto &s: strs) {
			int c = 0;
			for (auto &x: s) {
				if ((x - '0') % 2 == 1) {
					c ++;
				}
			}
			cnt.emplace_back(c);
		}

		sort(order.begin(), order.end(), [&] (int i, int j) {return cnt[i] > cnt[j];});

		for (int i = k; i < n; i ++) {
			if (cnt[order[i]] == cnt[order[k - 1]]) {
				for (auto &x: strs[order[i]]) {
					if ((x - '0') % 4 == 0) {
						swap(order[i], order[k - 1]);
						break;
					}
				}
			}
		}

		string tmp;
		char ch = '.';
		vector<int> idxs;

		for (int i = 0; i < k; i ++) {
			idxs.emplace_back(order[i]);
			for (auto &x: strs[order[i]]) {
				if ((x - '0') % 2 == 1) {
					tmp += x;
				}
				else if ((x - '0') % 4 == 0) {
					ch = x;
				}
			}
		}


		if (tmp.size() && ch != '.') tmp += ch;

		if (tmp.size() > ans.size()) {
			ans = tmp;
			chosen = idxs;
		}
	}

	if (ans.size()) {
		for (auto &x: chosen) cout << x + 1 << ' '; cout << '\n';
		cout << ans << '\n';
	}
	else {
		for (int i = 1; i <= k; i ++) cout << i << ' '; cout << '\n';
		cout << "empty\n";
	}

	return 0;
}
```
