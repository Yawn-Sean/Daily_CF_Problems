**提示 1：** 直接考虑每次变化结果是啥太复杂了。考虑对于某个字符串最少要操作几次。

先看提示。

对于某个目标字符串，最少操作几次呢？

一旦你最后某次进行了第二类操作，你就不如开始就做第二类操作，再调整到目标；第三类操作也是一样。

因此操作次数等于 $\min(直接进行第一类操作的次数, 1+用一次第二类操作后第一类操作的使用次数, 1+用一次第三类操作后第一类操作的使用次数)$ 。

所以我们直接枚举前 $i$ 项用三种方式分别操作次数为 $x,y,z$ 的情况的字符串数量为 $dp[i][x][y][z]$ ，这样转移时，只需枚举新字符是 $0$ 还是 $1$ 即可快速发现新的 $x',y',z'$。

最后提取时，只需满足三种操作的最小者总操作次数不超过 $k$ 即可。

这里可以使用滚动数组压缩空间。

时间复杂度为 $\mathcal{O}(n^4)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n, k = MII()
    s = [int(c) for c in I()]
    s1 = sorted(s)
    s2 = sorted(s, reverse=True)
    
    mod = 998244353
    
    def f(x, y, z): return (x * (n + 1) + y) * (n + 1) + z
    
    M = (n + 1) * (n + 1) * (n + 1)
    dp = [0] * M
    ndp = [0] * M
    
    dp[0] = 1
    
    cnt = 0
    for i in range(n):
        step = i
        for x in range(step + 1):
            for y in range(step + 1):
                for z in range(step + 1):
                    if dp[f(x, y, z)]:
                        for d in range(2):
                            nx = x + (s[i] != d)
                            ny = y + (s1[i] != d)
                            nz = z + (s2[i] != d)
                            ndp[f(nx, ny, nz)] += dp[f(x, y, z)]
        
        step = i + 1
        for x in range(step + 1):
            for y in range(step + 1):
                for z in range(step + 1):
                    dp[f(x, y, z)] = ndp[f(x, y, z)] % mod
                    ndp[f(x, y, z)] = 0
    
    ans = 0
    for x in range(n + 1):
        for y in range(n + 1):
            for z in range(n + 1):
                if x <= k or y < k or z < k:
                    ans += dp[f(x, y, z)]
    
    print(ans % mod)
```

C++ 做法如下——

```cpp []
const int N = 151, mod = 998244353;
int dp[N][N][N], ndp[N][N][N];

void add(int &x, int y) {
	x += y;
	if (x >= mod) x -= mod;
}

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int n, k;
	cin >> n >> k;

	string s, s1, s2, choice = "01";
	cin >> s;
	s1 = s, sort(s1.begin(), s1.end());
	s2 = s, sort(s2.rbegin(), s2.rend());

	dp[0][0][0] = 1;

	for (int i = 0; i < n; i ++) {
		for (int x = 0; x <= i; x ++) {
			for (int y = 0; y <= i; y ++) {
				for (int z = 0; z <= i; z ++) {
					if (dp[x][y][z]) {
						for (auto &c: choice) {
							int nx = x + (c != s[i]);
							int ny = y + (c != s1[i]);
							int nz = z + (c != s2[i]);
							add(ndp[nx][ny][nz], dp[x][y][z]);
						}
					}
				}
			}
		}

		for (int x = 0; x <= i + 1; x ++) {
			for (int y = 0; y <= i + 1; y ++) {
				for (int z = 0; z <= i + 1; z ++) {
					dp[x][y][z] = ndp[x][y][z];
					ndp[x][y][z] = 0;
				}
			}
		}
	}

	int ans = 0;

	for (int x = 0; x <= n; x ++) {
		for (int y = 0; y <= n; y ++) {
			for (int z = 0; z <= n; z ++) {
				if (x <= k || y < k || z < k) {
					add(ans, dp[x][y][z]);
				}
			}
		}
	}

	cout << ans;

	return 0;
}
```
