**提示 1：** 相当于奇数步和偶数步分别分配给 AD 和 BC 。

看提示。

奇数步一定在 AD 之间选，偶数步一定在 BC 之间选。

而我们只需要分配这些顺序，于是用组合数求再乘起来就行。

注意 $a\geq 2$ ，同时 $(a+d)-(b+c)$ 只可能是 $0$ 或者 $1$ ，不然输出 $0$ 就好。

时间复杂度为 $\mathcal{O}(1)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main(): 
    mod = 998244353
    M = 10 ** 6
    f = Factorial(M, mod)
    
    t = II()
    outs = []
    
    for _ in range(t):
        a, b, c, d = MII()
        
        if a + d - b - c not in {0, 1} or a == 1: outs.append('0')
        else:
            ans = f.combi(a + d - 2, a - 2) * f.combi(b + c, b)
            if a + d == b + c: ans *= 2
            outs.append(f'{ans % mod}')
    
    print('\n'.join(outs))
```

C++ 做法如下——

```cpp []
using mint = atcoder::modint998244353;

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int M = 1e6;
	vector<mint> f(M + 1), g(M + 1);

	f[0] = 1;
	for (int i = 1; i <= M; i ++) f[i] = i * f[i - 1];
	
	g[M] = 1 / f[M];
	for (int i = M; i >= 1; i --) g[i - 1] = i * g[i];

	auto comb = [&] (int x, int y) -> mint {
		if (y < 0 || y > x) return mint(0);
		return f[x] * g[y] * g[x - y];
	};

	int t;
	cin >> t;

	while (t --) {
		int a, b, c, d;
		cin >> a >> b >> c >> d;

		mint ans = comb(a + d - 2, a - 2) * comb(b + c, b);
		if (a + d - b - c == 1) ;
		else if (a + d - b - c == 0) ans *= 2;
		else ans = 0;

		cout << ans.val() << '\n';
	}

	return 0;
}
```
