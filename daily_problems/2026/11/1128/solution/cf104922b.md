**提示 1：** 一个区间异或后长啥样？

**提示 2：** 线段树上，这些区间？

一个区间在被一个数 $x$ 异或后，会变成 $\log 区间长度$ 量级的区间段。

同时，这些区间在线段树上的每一层中，需要调整的位置是连续的。

所以直接用懒更新线段树维护就是可行的了，尽管要更新 $\mathcal{O}(n)$ 个区间，但整体调整的线段树结点数也是 $\mathcal{O}(n)$ 的。当然每个结点只能是 $2^i$ 的长度。

时间复杂度为 $\mathcal{O}(2^n+nq)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main(): 
    n = II()
    
    N = 1 << n
    
    seg = [0] * N * 2
    lazy = [-1] * N
    
    for i, v in enumerate(MII()):
        seg[i + N] = v
    
    for i in range(N - 1, 0, -1):
        seg[i] = seg[i * 2] + seg[i * 2 + 1]
    
    q = II()
    outs = []
    
    def pushdown(pos):
        if lazy[pos] == -1: return
        
        w = lazy[pos] * (1 << n - pos.bit_length())
        seg[2 * pos] = w
        seg[2 * pos + 1] = w
        
        if pos * 2 < N:
            lazy[2 * pos] = lazy[pos]
            lazy[2 * pos + 1] = lazy[pos]
        
        lazy[pos] = -1
    
    def apply(pos, v):
        seg[pos] = v << n - pos.bit_length() + 1
        if pos < N: lazy[pos] = v
    
    def pushup(pos):
        if pos: seg[pos] = seg[2 * pos] + seg[2 * pos + 1]
    
    for _ in range(q):
        query = LII()
        
        if query[0] == 1:
            l, r, k, v = query[1:]
            
            l += N
            r += N + 1
            l0, r0 = l, r
    
            for i in range(n, 0, -1):
                if ((l0 >> i) << i) != l0:
                    pushdown((l0 >> i) ^ (k >> i))
                if ((r0 >> i) << i) != r0:
                    pushdown(((r0 - 1) >> i) ^ (k >> i))
    
            for i in range(n + 1):
                if l == r:
                    break
                if l & 1:
                    apply(l ^ (k >> i), v)
                    l += 1
                if r & 1:
                    r -= 1
                    apply(r ^ (k >> i), v)
                l >>= 1
                r >>= 1
    
            for i in range(1, n + 1):
                if ((l0 >> i) << i) != l0:
                    pushup((l0 >> i) ^ (k >> i))
                if ((r0 >> i) << i) != r0:
                    pushup(((r0 - 1) >> i) ^ (k >> i))
                
        else:
            l, r, k = query[1:]
            
            ans = 0
            l += N
            r += N + 1
            l0, r0 = l, r
    
            for i in range(n, 0, -1):
                if ((l0 >> i) << i) != l0:
                    pushdown((l0 >> i) ^ (k >> i))
                if ((r0 >> i) << i) != r0:
                    pushdown(((r0 - 1) >> i) ^ (k >> i))
    
            for i in range(n + 1):
                if l == r:
                    break
                if l & 1:
                    ans += seg[l ^ (k >> i)]
                    l += 1
                if r & 1:
                    r -= 1
                    ans += seg[r ^ (k >> i)]
                l >>= 1
                r >>= 1
    
            outs.append(ans)
    
    print('\n'.join(map(str, outs)))
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int n;
	cin >> n;

	int N = 1 << n;

	vector<long long> seg(N * 2);
	vector<int> lazy(N, -1);

	for (int i = N; i < N * 2; i ++) {
		int x; cin >> x;
		seg[i] = x;
	}

	for (int i = N - 1; i > 0; i --) {
		seg[i] = seg[i * 2] + seg[i * 2 + 1];
	}

	int q;
	cin >> q;

	auto bit_length = [&] (int x) -> int {
		return 32 - __builtin_clz(x);
	};

	auto pushdown = [&] (int pos) -> void {
		if (lazy[pos] == -1) return ;

		long long w = 1ll * lazy[pos] * (1 << n - bit_length(pos));

		seg[2 * pos] = w;
		seg[2 * pos + 1] = w;

		if (pos * 2 < N) {
			lazy[2 * pos] = lazy[pos];
			lazy[2 * pos + 1] = lazy[pos];
		}

		lazy[pos] = -1;
	};

	auto apply = [&] (int pos, int v) -> void {
		seg[pos] = 1ll * v * (1 << n - bit_length(pos) + 1);
		if (pos < N) lazy[pos] = v;
	};

	auto pushup = [&] (int pos) -> void {
		seg[pos] = seg[pos * 2] + seg[pos * 2 + 1];
	};

	while (q --) {
		int op;
		cin >> op;

		if (op == 1) {
			int l, r, k, v;
			cin >> l >> r >> k >> v;

			l += N;
			r += N + 1;

			int l0 = l, r0 = r;

			for (int i = n; i > 0; i --) {
				if (((l0 >> i) << i) != l0) pushdown((l0 >> i) ^ (k >> i));
				if (((r0 >> i) << i) != r0) pushdown(((r0 - 1) >> i) ^ (k >> i));
			}

			for (int i = 0; i <= n && l != r; i ++) {
				if (l & 1) apply(l ^ (k >> i), v), l ++;
				if (r & 1) r --, apply(r ^ (k >> i), v);
				l >>= 1;
				r >>= 1;
			}

			for (int i = 1; i <= n; i ++) {
				if (((l0 >> i) << i) != l0) pushup((l0 >> i) ^ (k >> i));
				if (((r0 >> i) << i) != r0) pushup(((r0 - 1) >> i) ^ (k >> i));
			}
		}
		else {
			int l, r, k;
			cin >> l >> r >> k;

			l += N;
			r += N + 1;

			long long ans = 0;

			for (int i = n; i > 0; i --) {
				if (((l >> i) << i) != l) pushdown((l >> i) ^ (k >> i));
				if (((r >> i) << i) != r) pushdown(((r - 1) >> i) ^ (k >> i));
			}

			for (int i = 0; i <= n && l != r; i ++) {
				if (l & 1) ans += seg[l ^ (k >> i)], l ++;
				if (r & 1) r --, ans += seg[r ^ (k >> i)];
				l >>= 1;
				r >>= 1;
			}

			cout << ans << '\n';
		}
	}

	return 0;
}
```
