**提示 1：** 那些不动的字符，贡献如何？

先考虑把所有字符先对齐。此时你先提取每个字符的位置。如果某个字符在两个列表中出现的次数不同，则显然无法构造；否则，应该前配前，后配后，直接计算成本。

那些不动的字符可以多省下一单位成本，因为不需要操作。而这些字符一定是原字符串的子序列，因此分别求每个位置的最长公共子序列即可。

注意这里，将同个位置的两个字符相互匹配，不会使得整体成本增加，因此先贪心匹配同个位置的最长公共子序列是不劣的。

时间复杂度为 $\mathcal{O}(n\times len^2+n|\Sigma|)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n = II()
    
    strs1 = LI()
    strs2 = LI()
    
    pos1 = [[] for _ in range(26)]
    pos2 = [[] for _ in range(26)]
    
    ans = 0
    
    for i in range(n):
        s1 = [ord(x) - ord('a') for x in strs1[i]]
        s2 = [ord(x) - ord('a') for x in strs2[i]]
        
        k1 = len(s1)
        k2 = len(s2)
        
        for j in range(k1):
            pos1[s1[j]].append(i)
        
        for j in range(k2):
            pos2[s2[j]].append(i)
        
        dp = [[0] * (k2 + 1) for _ in range(k1 + 1)]
        
        for i in range(k1):
            for j in range(k2):
                if s1[i] == s2[j]:
                    dp[i + 1][j + 1] = dp[i][j] + 1
                else:
                    dp[i + 1][j + 1] = fmax(dp[i][j + 1], dp[i + 1][j])
        
        ans -= dp[k1][k2]
    
    for i in range(26):
        if len(pos1[i]) != len(pos2[i]): exit(print(-1))
        else:
            for x, y in zip(pos1[i], pos2[i]):
                ans += abs(x - y) + 1
    
    print(ans)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int n;
	cin >> n;

	vector<string> strs1(n), strs2(n);
	for (auto &s: strs1) cin >> s;
	for (auto &s: strs2) cin >> s;

	vector<vector<int>> pos1(26), pos2(26);

	long long ans = 0;

	for (int i = 0; i < n; i ++) {
		auto &s1 = strs1[i];
		auto &s2 = strs2[i];

		int k1 = s1.size();
		int k2 = s2.size();

		for (auto &c: s1) pos1[c - 'a'].emplace_back(i);
		for (auto &c: s2) pos2[c - 'a'].emplace_back(i);

		vector<vector<int>> dp(k1 + 1, vector<int>(k2 + 1, 0));

		for (int x = 0; x < k1; x ++) {
			for (int y = 0; y < k2; y ++) {
				if (s1[x] == s2[y]) dp[x + 1][y + 1] = dp[x][y] + 1;
				else dp[x + 1][y + 1] = max(dp[x][y + 1], dp[x + 1][y]);
			}
		}

		ans -= dp[k1][k2];
	}

	for (int i = 0; i < 26; i ++) {
		if (pos1[i].size() != pos2[i].size()) return cout << -1, 0;

		for (int j = 0; j < pos1[i].size(); j ++) {
			ans += abs(pos1[i][j] - pos2[i][j]) + 1;
		}
	}

	cout << ans;

	return 0;
}
```
