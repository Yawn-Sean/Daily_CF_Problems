**提示 1：** 如何选出作为答案的活动的下标集合？

因为越靠后的比赛的自由度越低，所以从后往前遍历。假设当前维护了一个需要选取的下标集合。

则遍历到某个位置时，先加入当前的活动，此时这个后缀可选的是当前集合的活动。一旦你发现，此时的下标集合放不下这么多活动，你就需要删去活动中最不值的那一个，这样贪心一定使得总价值最大。

而选好下标后，只需将所有活动的时间尽可能往后推即可，我们只需让每个活动后剩余的天数能放下此后算分的活动的个数即可。

时间复杂度为 $\mathcal{O}(n\log n)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n, D, t = MII()
    
    ds = []
    bs = []
    
    for _ in range(n):
        d, b = MII()
        ds.append(d)
        bs.append(b)
    
    to_use = [1] * n
    pq = []
    
    for i in range(n - 1, -1, -1):
        d = ds[i]
        b = bs[i]
        
        heappush(pq, bs[i] * n + i)
        
        while D - d + 1 < len(pq):
            to_use[heappop(pq) % n] = 0
    
    ans = 0
    for i in range(n):
        if to_use[i]:
            ans += bs[i]
    
    print(ans)
    
    if t == 2:
        for i in range(n - 1, 0, -1):
            to_use[i - 1] += to_use[i]
        
        for i in range(n):
            to_use[i] = fmin(D - to_use[i] + 1, D)
        
        print(' '.join(map(str, to_use)))
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int n, D, t;
	cin >> n >> D >> t;

	vector<pair<int, int>> items(n);
	for (auto &[x, y]: items) cin >> x >> y;

	vector<int> to_use(n, 1);
	priority_queue<pair<int, int>> pq;

	for (int i = n - 1; i >= 0; i --) {
		auto [x, y] = items[i];
		pq.push({-y, i});

		while (D - x + 1 < pq.size()) {
			to_use[pq.top().second] = 0;
			pq.pop();
		}
	}

	long long ans = 0;

	for (int i = 0; i < n; i ++) {
		if (to_use[i]) {
			ans += items[i].second;
		}
	}

	cout << ans << '\n';

	if (t == 2) {
		for (int i = n - 1; i > 0; i --) to_use[i - 1] += to_use[i];
		for (int i = 0; i < n; i ++) to_use[i] = min(D - to_use[i] + 1, D);
		for (int i = 0; i < n; i ++) cout << to_use[i] << " \n"[i == n - 1];
	}

	return 0;
}
```
