**提示 1：** 答案相当于所有情况的所有颜色的和。交换求和顺序。

**提示 2：** 假设你获取了某个时刻的 Trie，如何求解答案？

先看提示 1 。于是考虑每个位置对结果的贡献。

此时类似于一个扫描线，你维护当前的修改操作集合，在 $L$ 处新增，在 $R+1$ 处删除。我们只需考虑这些操作在不同异或操作后，最后出现的颜色的和。

此时经过异或操作，所以考虑把整个情况放到一个 Trie 里。在 Trie 的 $d$ 位置更新或撤销 $c$ 的数值。

在这个 Trie 上，我们尝试更新整体的计算结果。

假设考虑最低的 $i$ 位时，异或上 $0\sim 2^i-1$ 的数的结果之和是 $dp[x]$ 。

则其两个子树如果非空，则在异或后，哪个子树都可能排在后面，所以 DP 数值等于两者之和。

如果只有一棵子树，则结果翻倍。

如果没有子树，结果是 $0$ 。

需要判子树是否为空，所以比较方便的做法是顺便维护下 Trie 上每个结点对应子树的大小。

在插入元素和删除元素时，只有插入和删除的这条链上的各个信息需要更新，因此复杂度是 $\mathcal{O}(k)$ 的。

时间复杂度为 $\mathcal{O}(n+qk)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main(): 
    t = II()
    outs = []
    
    mod = 10 ** 9 + 7
    
    for _ in range(t):
        n, k, q = MII()
        updates = [[] for _ in range(n + 1)]
        
        for _ in range(q):
            d, l, r, c = MII()
            updates[l - 1].append((1, d, c))
            updates[r].append((-1, d, c))
        
        trie01 = [[-1] * (q * k + 1) for _ in range(2)]
        sz = [0] * (q * k + 1)
        dp = [0] * (q * k + 1)
        
        ans = 0
        pt = 1
        
        for i in range(n):
            for op, d, c in updates[i]:
                cur = 0
                stk = []
                
                for bit in range(k - 1, -1, -1):
                    w = d >> bit & 1
                    stk.append(cur)
                    
                    if trie01[w][cur] == -1:
                        trie01[w][cur] = pt
                        pt += 1
                    
                    cur = trie01[w][cur]
                    sz[cur] += op
                
                dp[cur] += c * op
                
                for cur in reversed(stk):
                    v0 = trie01[0][cur]
                    v1 = trie01[1][cur]
                    
                    f0 = v0 >= 0 and sz[v0] > 0
                    f1 = v1 >= 0 and sz[v1] > 0
                    
                    if f0:
                        if f1: dp[cur] = (dp[v0] + dp[v1]) % mod
                        else: dp[cur] = dp[v0] * 2 % mod
                    else:
                        if f1: dp[cur] = dp[v1] * 2 % mod
                        else: dp[cur] = 0
                
            ans += dp[0]
            ans %= mod
        
        outs.append(ans)
    
    print('\n'.join(map(str, outs)))
```

C++ 做法如下——

```cpp []
#include "atcoder/modint.hpp"
using mint = atcoder::modint1000000007;

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int t;
	cin >> t;

	while (t --) {
		int n, k, q;
		cin >> n >> k >> q;

		vector<vector<tuple<int, long long, int>>> updates(n + 1);

		for (int i = 0; i < q; i ++) {
			long long d;
			int l, r, c;
			cin >> d >> l >> r >> c;
			updates[l - 1].push_back({1, d, c});
			updates[r].push_back({-1, d, c});
		}

		mint ans = 0;

		vector<vector<int>> trie(2, vector<int>(q * k + 5, -1));
		vector<int> sz(q * k + 5);
		vector<mint> dp(q * k + 5);

		int pt = 1;

		for (int i = 0; i < n; i ++) {
			for (auto &[op, d, c]: updates[i]) {
				int cur = 0;
				vector<int> stk;

				for (int bit = k - 1; bit >= 0; bit --) {
					stk.emplace_back(cur);
					int w = d >> bit & 1;
					if (trie[w][cur] == -1) trie[w][cur] = pt ++;
					cur = trie[w][cur];
					sz[cur] += op;
				}

				dp[cur] += c * op;

				reverse(stk.begin(), stk.end());

				for (auto &idx: stk) {
					int v0 = trie[0][idx];
					int v1 = trie[1][idx];

					bool f0 = v0 >= 0 && sz[v0] > 0;
					bool f1 = v1 >= 0 && sz[v1] > 0;

					if (f0) {
						if (f1) dp[idx] = dp[v0] + dp[v1];
						else dp[idx] = 2 * dp[v0];
					}
					else {
						if (f1) dp[idx] = 2 * dp[v1];
						else dp[idx] = 0;
					}
				}
			}

			ans += dp[0];
		}

		cout << ans.val() << '\n';
	}

	return 0;
}
```
