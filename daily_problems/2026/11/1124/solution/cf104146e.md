**提示 1：** 每个砝码对称量物品的权重如何？

**提示 2：** 如果要严格表示重量？

**提示 3：** 其实不等关系也足够确定答案了。

考虑每个物品可以放在物品同一侧或者另一侧，或者干脆不用。而我们相当于做 $w_0+w_{i_1}+w_{i_2}+\dots$ 和 $w_{j_1}+w_{j_2}+\dots$ 的比较，这也就等价于给每个物品赋予权重 $1,0,-1$ ，再跟待测物品比较。

如果我们要完全凑出物体的质量，则 $1$ 是必须的。第二个是 $3$ ，因为 $3-1$ 足够测出 $2$ 。此时能测出的物品重量是 $1\sim 4$ ，所以下一个物品可以选择 $4\times 2+1$ 。类似不断往后推。

这样最后一个砝码是 $3^{14}$ ，此时砝码总重并不够。

怎么办呢？

我们发现每个砝码的质量都可以乘 $2$ ，因为我们能精确测出所有偶数质量的情况下，如果有个物品质量在两个相邻的偶数之间，那么我们也可以断定那个物品的重量了。

所以我们只需在所有偶数中二分，如果中途搜到了就确定了重量；否则一定得到某个区间，直接取区间内的奇数即可。

Python 做法如下——

```Python []
def main(): 
    nums = [2 * 3 ** i for i in range(15)]
    
    print(15)
    print(*nums)
    print(sum(nums))
    sys.stdout.flush()
    
    def check(mid):
        pos = []
        neg = [0]
        
        for i in range(15):
            if mid % 3 == 1:
                mid -= 1
                pos.append(i + 1)
            elif mid % 3 == 2:
                mid += 1
                neg.append(i + 1)
            mid //= 3
        
        print('WEIGH')
        print(len(neg))
        print(*neg)
        print(len(pos))
        print(*pos)
        sys.stdout.flush()
        
        s = I()
        if s == '<': return -1
        if s == '>': return 1
        return 0
    
    t = II()
    bound = sum(nums) // 2
    
    for _ in range(t):
        l, r = 1, bound
        
        ans = -1
        
        while l <= r:
            mid = (l + r) // 2
            res = check(mid)
            
            if res == -1: r = mid - 1
            elif res == 1: l = mid + 1
            else:
                ans = mid * 2
                break
        
        print('VERDICT')
        if ans != -1:
            print('EXACT')
            print(ans, flush=True)
        else:
            if l > bound:
                print('HEAVY', flush=True)
            else:
                print('EXACT')
                print(l + r, flush=True)
        I()
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	cout << 15 << '\n';
	int cur = 2, total = 0;
	for (int i = 0; i < 15; i ++) {
		cout << cur << " \n"[i == 14];
		total += cur;
		cur *= 3;
	}

	cout << total << endl;

	auto query = [&] (int x) -> int {
		vector<int> pos, neg = {0};
		for (int i = 1; i <= 15; i ++) {
			if (x % 3 == 1) pos.emplace_back(i), x --;
			else if (x % 3 == 2) neg.emplace_back(i), x ++;
			x /= 3;
		}

		cout << "WEIGH\n";
		cout << neg.size() << '\n';
		for (int i = 0; i < neg.size(); i ++) cout << neg[i] << " \n"[i + 1 == neg.size()];
		cout << pos.size() << '\n';
		for (int i = 0; i < pos.size(); i ++) cout << pos[i] << " \n"[i + 1 == pos.size()];
		cout.flush();

		char c;
		cin >> c;

		if (c == '<') return -1;
		if (c == '>') return 1;
		return 0;
	};

	int t;
	cin >> t;

	while (t --) {
		int l = 1, r = total / 2;
		int ans = -1;

		while (l <= r) {
			int mid = (l + r) / 2;
			int res = query(mid);

			if (res == -1) r = mid - 1;
			else if (res == 1) l = mid + 1;
			else {
				ans = mid * 2;
				break;
			}
		}

		if (ans != -1) {
			cout << "VERDICT\n";
			cout << "EXACT\n";
			cout << ans << endl;
		}
		else {
			if (l > total / 2) {
				cout << "VERDICT\n";
				cout << "HEAVY" << endl;
			}
			else {
				cout << "VERDICT\n";
				cout << "EXACT\n";
				cout << l + r << endl;
			}
		}

		string s;
		cin >> s;
	}

	return 0;
}
```
