**提示 1：** 不断推断。最后剩下什么？

可以先把所有写了数字的位置放入栈中，接下来用类似拓扑排序的方法：

如果某个位置周围只有 $k$ 个不知道有没有雷的位置，但结果刚好是 $0$ 个 / $k$ 个雷，我们就可以进行推断了。

同时把这些位置的相邻位置加入栈中，就可以进行进一步的推断了。

就这样不断往复。最后剩下的只可能是一堆没有办法推断的部分，也就完成了任务。

时间复杂度为 $\mathcal{O}(n)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n = II()
    s = [int(c) if c != '?' else -1 for c in I()]
    
    ans = 0
    stk = [i for i in range(n) if s[i] != -1]
    
    while stk:
        x = stk.pop()
        
        if s[x] >= 0:
            total = s[x]
            tmp = []
            
            for dx in [-1, 1]:
                if 0 <= x + dx < n:
                    if s[x + dx] == -1:
                        tmp.append(x + dx)
                    elif s[x + dx] == -2:
                        total -= 1
    
            if total == 0:
                for u in tmp:
                    ans += 1
                    s[u] = -3
                    stk.append(u)
            
            elif total == len(tmp):
                for u in tmp:
                    s[u] = -2
                    stk.append(u)
        else:
            saved_x = x
            for vx in [-1, 1]:
                x = saved_x + vx
                if 0 <= x < n:
                    total = s[x]
                    tmp = []
                    
                    for dx in [-1, 1]:
                        if 0 <= x + dx < n:
                            if s[x + dx] == -1:
                                tmp.append(x + dx)
                            elif s[x + dx] == -2:
                                total -= 1
    
                    if total == 0:
                        for u in tmp:
                            ans += 1
                            s[u] = -3
                            stk.append(u)
                    elif total == len(tmp):
                        for u in tmp:
                            s[u] = -2
                            stk.append(u)
    
    print(ans)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int n;
	string s;

	cin >> n >> s;

	vector<int> stk;

	for (int i = 0; i < n; i ++) {
		if (s[i] >= '0' && s[i] <= '2') {
			stk.emplace_back(i);
		}
	}

	int ans = 0;

	while (!stk.empty()) {
		int u = stk.back(); stk.pop_back();

		vector<int> dirs;
		if (s[u] >= '0' && s[u] <= '2') dirs = {0};
		else dirs = {-1, 1};

		for (auto &nx: dirs) {
			int x = u + nx;

			if (x >= 0 && x < n && s[x] >= '0' && s[x] <= '2') {
				int total = s[x] - '0';
				vector<int> tmp;

				for (auto &dx: {-1, 1}) {
					if (x + dx >= 0 && x + dx < n) {
						if (s[x + dx] == '*') total --;
						else if (s[x + dx] == '?') tmp.emplace_back(x + dx);
					}
				}

				if (total == 0) {
					for (auto &v: tmp) {
						s[v] = ' ';
						ans ++;
						stk.emplace_back(v);
					}
				}
				else if (total == tmp.size()) {
					for (auto &v: tmp) {
						s[v] = '*';
						stk.emplace_back(v);
					}
				}
			}
		}
	}

	cout << ans;

	return 0;
}
```
