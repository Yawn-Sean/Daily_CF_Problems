**提示 1：** 依次确定每个数值。如何判断一个区间内是否有 $x$ ？

我们可以用 $0,1,\dots,x-1$ 的集合 $Q$ 和任何一个下标子集的查询，判断这些下标中是否存在 $x$ 。

所以，对于每一个 $x=0,1,\dots,n$ ，我们都可以独立查其中的 $x$ 。

具体而言，首先查整体是否有 $x$ ；如果没有，查下一个数 $x + 1$ ，否则可以二分找到第一个 $x$ ，再对剩余的下标继续确认是否有 $x$ ，有 $x$ 就继续二分，如是往复。

这样查询多少次呢？每个位置的数值需要 $\mathcal{O}(\log n)$ 次查询，同时 $0\sim n$ 都会多出一次废的不进行二分的查询，所以总查询次数大概是 $n\log n + n$ ，所以给定的查询次数相当充裕。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    def query(p, q):
        print('?', len(p), len(q))
        print(' '.join(f'{x + 1}' for x in p))
        print(' '.join(map(str, q)))
        sys.stdout.flush()
        return II()
    
    def answer(lst):
        print('!', ' '.join(map(str, lst)), flush=True)
    
    t = II()
    
    for _ in range(t):
        n = II()
        ans = [-1] * n
        
        for val in range(n + 1):
            idxs = [i for i in range(n) if ans[i] == -1]
            if len(idxs) == 0: break
            
            q = list(range(val))
            
            pt = 0
            while pt < len(idxs):
                if query(idxs[pt:], q) == val: break
                
                left, right = pt, len(idxs) - 1
                while left <= right:
                    mid = (left + right) // 2
                    if query(idxs[pt:mid + 1], q) == val: left = mid + 1
                    else: right = mid - 1
                
                ans[idxs[left]] = val
                pt = left + 1
        
        answer(ans)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int t;
	cin >> t;

	while (t --) {
		int n;
		cin >> n;

		vector<int> ans(n, -1);

		for (int val = 0; val <= n; val ++) {
			vector<int> idxs;
			for (int i = 0; i < n; i ++) {
				if (ans[i] == -1) {
					idxs.emplace_back(i);
				}
			}

			int pt = 0, query_res;

			while (pt < idxs.size()) {
				cout << "? " << idxs.size() - pt << ' ' << val << '\n';
				for (int i = pt; i < idxs.size(); i ++) cout << idxs[i] + 1 << ' ';
				cout << '\n';
				for (int i = 0; i < val; i ++) cout << i << ' ';
				cout << endl;
				cin >> query_res;
				if (query_res == val) break;

				int left = pt, right = idxs.size() - 1;
				while (left <= right) {
					int mid = (left + right) / 2;
					cout << "? " << mid - pt + 1 << ' ' << val << '\n';
					for (int i = pt; i <= mid; i ++) cout << idxs[i] + 1 << ' ';
					cout << '\n';
					for (int i = 0; i < val; i ++) cout << i << ' ';
					cout << endl;
					cin >> query_res;

					if (query_res == val) left = mid + 1;
					else right = mid - 1;
				}

				ans[idxs[left]] = val;
				pt = left + 1;
			}
		}

		cout << "!";
		for (auto &x: ans) cout << ' ' << x;
		cout << endl;
	}

	return 0;
}
```
