**提示 2：** 不能猜的时候咋办？

每次查询都会告诉你是距离多少——你就可以猜是在左边还是右边了，你随机猜的话会有 $\frac{1}{2}$ 的概率猜对。

这样，到 $30$ 次，大概率就解决了。

但还有问题——如果只剩下一个雷了，你就不能随便猜了。

此时最坏的情况只有两个被选位置，总能查询某个位置，使得你查询它能确定最终答案是哪个位置。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    def query(x):
        global n
        print('?', x, flush=True)
        ret = LI()
        if ret[0] == 'YES': n -= 1
        return (ret[0] == 'YES'), int(ret[1])
    
    def answer(x):
        print('!', x)
        exit()
    
    def dis(x, y):
        w = abs(x - y)
        return fmin(w, L - w)
    
    n = II()
    L = 10 ** 9
    
    cur = random.randint(1, L)
    flg, d = query(cur)
    
    while n > 1:
        x = (cur - d - 1) % L + 1
        y = (cur + d - 1) % L + 1
        
        V = x if random.randint(0, 1) else y
        flg, d = query(V)
        
        if not flg: answer(x ^ y ^ V)
        cur = V
    
    x = (cur - d - 1) % L + 1
    y = (cur + d - 1) % L + 1
    
    if x == y: answer(x)
    else:
        for i in range(1, L + 1):
            if x != i and y != i and dis(i, x) != dis(i, y):
                if query(i)[1] == dis(i, x): answer(x)
                else: answer(y)
```

C++ 做法如下——

```cpp []
mt19937 rng{random_device{}()};

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int L = 1e9;

	uniform_int_distribution<int> rng_L(1, L);
	uniform_int_distribution<int> rng_1(0, 1);

	int n;
	cin >> n;

	auto query = [&] (int x) -> pair<bool, int> {
		cout << "? " << x << endl;
		string s; int d;
		cin >> s >> d;
		if (s == "YES") n --;
		return {s == "YES", d};
	};

	auto answer = [&] (int x) -> void {
		cout << "! " << x << endl;
		exit(0);
		return ;
	};

	int cur = rng_L(rng);
	auto [flg, d] = query(cur);

	while (n > 1) {
		int x = (cur - d) % L; x = (x + L - 1) % L + 1;
		int y = (cur + d) % L; y = (y + L - 1) % L + 1;

		int V = rng_1(rng) ? x : y;
		auto [nflg, nd] = query(V);
		flg = nflg, d = nd;

		if (!flg) answer(x ^ y ^ V);
		cur = V;
	}

	int x = (cur - d) % L; x = (x + L - 1) % L + 1;
	int y = (cur + d) % L; y = (y + L - 1) % L + 1;

	if (x == y) answer(x);
	else {
		auto dis = [&] (int x, int y) -> int {
			int w = abs(x - y);
			return min(w, L - w);
		};

		for (int i = 1; i <= L; i ++) {
			if (x != i && y != i && dis(i, x) != dis(i, y)) {
				if (query(i).second == dis(i, x)) answer(x);
				else answer(y);
			}
		}
	}

	return 0;
}
```
