**提示 1：** 如何判断一个硬币组合是否成立？

**提示 2：** 减少你要判的组合数量。

如果你已经确定了硬币集，你可以用背包 DP 看 $n/2$ 能否被凑出来。

接下来尝试构造集。

首先 $1,5,10,50,100$ 都没必要用超过 $2$ 个，不然替换成 $2,10,20,100,200$ 不会更劣。

同理， $2,20$ 没必要用超过 $4$ 个，不然替换成 $10,100$ 不会更劣。

因此我们枚举这里，然后最后分配 $200,500$ ，得到所有凑成 $n$ 的硬币集，分别进行背包 DP 即可。

这里可以加剪枝减少搜索空间，不过整体空间也不大，可以验证发现需要 DP 的硬币组合数量很有限。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n = II()
    
    bills    = [1, 2, 5, 10, 20, 50, 100, 500, 200]
    bounds   = [1, 4, 1,  1,  4,  1,   1,  20,  50]
    suff_gcd = [1, 1, 5, 10, 10, 50, 100, 100, 200]
    
    if n % 2:
        print(n)
        print(*[1] * n)
        exit()
    
    
    def search(cur_value, idx, tmp):
        if cur_value % suff_gcd[idx]:
            return 
        
        if idx == 8:
            saved = tmp[:] + [200] * (cur_value // 200)
            
            dp = [0] * (n // 2 + 1)
            dp[0] = 1
            
            for x in saved:
                for i in range(n // 2, x - 1, -1):
                    if dp[i - x]: dp[i] = 1
            
            if dp[n // 2] == 0:
                print(len(saved))
                print(*saved)
                exit()
            
            return 
    
        cnt = fmin(cur_value // bills[idx], bounds[idx])
        search(cur_value, idx + 1, tmp)
        
        for i in range(1, cnt + 1):
            tmp.append(bills[idx])
            search(cur_value - bills[idx] * i, idx + 1, tmp)
        
        for i in range(cnt):
            tmp.pop()
    
    search(n, 0, [])
    
    print('splittable')
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int n;
	cin >> n;

	vector<int> bills    = {1, 2, 5, 10, 20, 50, 100, 500, 200};
	vector<int> bounds   = {1, 4, 1,  1,  4,  1,   1,  20,  50};
	vector<int> suff_gcd = {1, 1, 5, 10, 10, 50, 100, 100, 200};

	if (n & 1) {
		cout << n << '\n';
		for (int i = 0; i < n; i ++) cout << 1 << " \n"[i == n - 1];
		return 0;
	}

	auto dfs = [&] (auto &self, int cur_value, int idx, vector<int> &tmp) -> void {
		if (cur_value % suff_gcd[idx]) return ;

		if (idx == 8) {
			auto saved = tmp;
			for (int i = 0; i < cur_value / 200; i ++) saved.emplace_back(200);

			vector<int> dp(n / 2 + 1, 0);
			dp[0] = 1;

			for (auto &x: saved) {
				for (int i = n / 2; i >= x; i --) {
					dp[i] |= dp[i - x];
				}
			}

			if (dp[n / 2] == 0) {
				int k = saved.size();

				cout << k << '\n';
				for (int i = 0; i < k; i ++) cout << saved[i] << " \n"[k == n - 1];

				exit(0);
			}

			return ;
		}

		int cnt = min(cur_value / bills[idx], bounds[idx]);
		self(self, cur_value, idx + 1, tmp);

		for (int i = 1; i <= cnt; i ++) {
			tmp.emplace_back(bills[idx]);
			self(self, cur_value - bills[idx] * i, idx + 1, tmp);
		}

		for (int i = 1; i <= cnt; i ++) tmp.pop_back();
	};

	vector<int> tmp;

	dfs(dfs, n, 0, tmp);

	cout << "splittable\n";

	return 0;
}
```
