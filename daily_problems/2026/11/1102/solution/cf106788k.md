**提示 1：** 每个矩形一定可以被 “大矩形” 包含。先用大矩形估计答案。

**提示 2：** 避开重叠。

任何一个不包含 $(0, 0)$ 的矩形一定在上下左右的其中一个方位。

所以我们先用这四个方向的大矩形，枚举子集，看至少需要多少个方向的。

确定选取的方案后，把矩形进行调整——重叠的部分删掉就行了。

时间复杂度为 $\mathcal{O}(n)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main(): 
    n = II()
    xs = []
    ys = []
    
    for _ in range(n):
        x, y = MII()
        xs.append(x)
        ys.append(y)
    
    tags = [(0, 1), (1, 0), (0, -1), (-1, 0)]
    
    ans = 5
    chosen_msk = 0
    
    for i in range(16):
        flg = True
        
        for j in range(n):
            nflg = False
            
            for k in range(4):
                if i >> k & 1:
                    dx, dy = tags[k]
                    if xs[j] * dx + ys[j] * dy > 0:
                        nflg = True
            
            if not nflg:
                flg = False
        
        if flg and i.bit_count() < ans:
            ans = i.bit_count()
            chosen_msk = i
    
    chosen = [chosen_msk >> i & 1 for i in range(4)]
    
    ans = []
    
    if chosen[0]:
        if chosen[3]:
            ans.append('-0.4 0.6 1500.0 0.6 -0.4 1500.0 1500.0 1500.0')
        else:
            ans.append('-1500.0 0.6 1500.0 0.6 -1500.0 1500.0 1500.0 1500.0')
    
    if chosen[1]:
        if chosen[0]:
            ans.append('0.6 0.4 1500.0 0.4 0.6 -1500.0 1500.0 -1500.0')
        else:
            ans.append('0.6 1500.0 1500.0 1500.0 0.6 -1500.0 1500.0 -1500.0')
    
    if chosen[2]:
        if chosen[1]:
            ans.append('-1500.0 -0.6 0.4 -0.6 -1500.0 -1500.0 0.4 -1500.0')
        else:
            ans.append('-1500.0 -0.6 1500.0 -0.6 -1500.0 -1500.0 1500.0 -1500.0')
    
    if chosen[3]:
        if chosen[2]:
            ans.append('-1500.0 -0.4 -0.6 -0.4 -1500.0 1500.0 -0.6 1500.0')
        else:
            ans.append('-1500.0 -1500.0 -0.6 -1500.0 -1500.0 1500.0 -0.6 1500.0')
    
    print(len(ans))
    print('\n'.join(ans))
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int n;
	cin >> n;

	vector<pair<int, int>> pts(n);
	for (auto &[x, y]: pts) cin >> x >> y;

	vector<pair<int, int>> tags = {{0, 1}, {1, 0}, {0, -1}, {-1, 0}};

	int best = 5;
	int chosen_msk = 0;

	for (int mask = 0; mask < 16; mask ++) {
		bool valid = true;

		for (int j = 0; j < n; j ++) {
			bool covered = false;
			auto [x, y] = pts[j];

			for (int k = 0; k < 4; k ++) {
				if ((mask >> k) & 1) {
					auto [dx, dy] = tags[k];
					if (x * dx + y * dy > 0) covered = true;
				}
			}

			if (!covered) valid = false;
		}

		if (valid && __popcount(mask) < best) {
			best = __popcount(mask);
			chosen_msk = mask;
		}
	}

	vector<int> chosen(4);
	for (int i = 0; i < 4; ++i) chosen[i] = (chosen_msk >> i) & 1;

	cout << best << '\n';

	if (chosen[0]) {
		if (chosen[3]) cout << "-0.4 0.6 1500.0 0.6 -0.4 1500.0 1500.0 1500.0\n";
		else cout << "-1500.0 0.6 1500.0 0.6 -1500.0 1500.0 1500.0 1500.0\n";
	}
	if (chosen[1]) {
		if (chosen[0]) cout << "0.6 0.4 1500.0 0.4 0.6 -1500.0 1500.0 -1500.0\n";
		else cout << "0.6 1500.0 1500.0 1500.0 0.6 -1500.0 1500.0 -1500.0\n";
	}
	if (chosen[2]) {
		if (chosen[1]) cout << "-1500.0 -0.6 0.4 -0.6 -1500.0 -1500.0 0.4 -1500.0\n";
		else cout << "-1500.0 -0.6 1500.0 -0.6 -1500.0 -1500.0 1500.0 -1500.0\n";
	}
	if (chosen[3]) {
		if (chosen[2]) cout << "-1500.0 -0.4 -0.6 -0.4 -1500.0 1500.0 -0.6 1500.0\n";
		else cout << "-1500.0 -1500.0 -0.6 -1500.0 -1500.0 1500.0 -0.6 1500.0\n";
	}

	return 0;
}
```
