**提示 1：** 考虑每个比赛维护一堆信息，再合并。

$[1,2^n]$ 的结果取决于两侧的结果，与 $d$ 的取值。

因此，考虑每一对要比赛的对手是哪个区间决胜过来的。此时每个区间的分数取决于左右两侧的胜者以及此时 $d$ 的取值。

因此考虑维护每一侧的 $d$ 在某个区间 $[l,r]$ 内，胜者是 $x$ ，且总分是 $s$ 。

则两侧区间有交时，会发生合并，此时新区间求交，总分求和再加上当前比赛的分数。

而胜者取决于 $d$ 的数值，因此还可能对 $[l,r]$ 区间进行拆分。

而左右两侧子树我们可以让维护的结构中 $d$ 的取值范围递增，这样我们就可以用双指针执行这样的合并了。

设长度为 $2^k$ 的区间产生的结果序列的数量级为 $T(k)$ ，则首先两侧 $T(k-1)$ 长度，双指针合并后为 $2T(k-1)$ ，同时 $d$ 可能将每个区间拆为两个，因此总序列长度 $T(k)=4T(k-1)$ 。所以 $T(k)$ 是 $\mathcal{O}(4^k)$ 量级的。

而我们的总复杂度等于我们构造过的序列的长度之和，第 $1$ 层构造 $4^n$ 长度的 $1$ 个序列，……，第 $i$ 层构造 $4^{n+1-i}$ 长度的 $2^i$ 个序列，因此总序列长度等于 $4^n(1+1/2+\dots)=\mathcal{O}(4^n)$ 。

时间复杂度为 $\mathcal{O}(4^n)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n = II()
    gs = LII()
    ms = LII()
    
    def f(l, r):
        if l == r: return [(0, 10 ** 9, l, 0)]
        
        mid = (l + r) // 2
        vl = f(l, mid)
        vr = f(mid + 1, r)
        
        pl, pr = 0, 0
        
        ans = []
        
        while pl < len(vl) and pr < len(vr):
            dl1, dr1, win1, score1 = vl[pl]
            dl2, dr2, win2, score2 = vr[pr]
            
            dl = fmax(dl1, dl2)
            dr = fmin(dr1, dr2)
            
            if dl <= dr:
                score = score1 + score2 + ms[win1] + ms[win2]
                
                if gs[win1] > gs[win2]:
                    sep = gs[win1] - gs[win2]
                    
                    if dl <= sep <= dr:
                        ans.append((dl, sep, win1, score))
                    
                    if dl <= sep + 1 <= dr:
                        ans.append((sep + 1, dr, win2, score))
                    
                    if sep > dr:
                        ans.append((dl, dr, win1, score))
                    
                    if sep + 1 < dl:
                        ans.append((dl, dr, win2, score))
    
                else:
                    sep = gs[win2] - gs[win1]
                    
                    if dl <= sep <= dr:
                        ans.append((dl, sep, win2, score))
                    
                    if dl <= sep + 1 <= dr:
                        ans.append((sep + 1, dr, win1, score))
                    
                    if sep > dr:
                        ans.append((dl, dr, win2, score))
                    
                    if sep + 1 < dl:
                        ans.append((dl, dr, win1, score))
                
            if dr1 < dr2: pl += 1
            else: pr += 1
        
        return ans
    
    chosen = -1
    ans = -1
    
    for dl, dr, win, score in f(0, (1 << n) - 1):
        if score > ans:
            chosen = dl
            ans = score
    
    print(ans)
    print(chosen)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int n;
	cin >> n;

	vector<int> gs(1 << n), ms(1 << n);
	for (auto &g: gs) cin >> g;
	for (auto &m: ms) cin >> m;

	int inf = 1e9;

	auto dfs = [&] (auto &self, int l, int r) -> vector<array<long long, 4>> {
		if (l == r) return {{0, inf, l, 0}};

		int mid = (l + r) / 2;
		auto vl = self(self, l, mid);
		auto vr = self(self, mid + 1, r);

		int pl = 0, pr = 0;

		vector<array<long long, 4>> ans;

		while (pl < vl.size() && pr < vr.size()) {
			auto [dl1, dr1, win1, score1] = vl[pl];
			auto [dl2, dr2, win2, score2] = vr[pr];

			int dl = max(dl1, dl2);
			int dr = min(dr1, dr2);

			if (dl <= dr) {
				long long score = score1 + score2 + ms[win1] + ms[win2];
			
				if (gs[win1] > gs[win2]) {
					int sep = gs[win1] - gs[win2];
					if (sep >= dl && sep <= dr) ans.push_back({dl, sep, win1, score});
					if (sep + 1 >= dl && sep + 1 <= dr) ans.push_back({sep +  1, dr, win2, score});
					if (sep > dr) ans.push_back({dl, dr, win1, score});
					if (sep + 1 < dl) ans.push_back({dl, dr, win2, score});
				}
				else {
					int sep = gs[win2] - gs[win1];
					if (sep >= dl && sep <= dr) ans.push_back({dl, sep, win2, score});
					if (sep + 1 >= dl && sep + 1 <= dr) ans.push_back({sep +  1, dr, win1, score});
					if (sep > dr) ans.push_back({dl, dr, win2, score});
					if (sep + 1 < dl) ans.push_back({dl, dr, win1, score});
				}
			}

			if (dr1 < dr2) pl ++;
			else pr ++;
		}

		return ans;
	};

	long long ans = -1;
	int chosen = -1;

	for (auto &[dl, dr, win, score]: dfs(dfs, 0, (1 << n) - 1)) {
		if (score > ans) {
			ans = score;
			chosen = dl;
		}
	}

	cout << ans << '\n' << chosen;

	return 0;
}
```
