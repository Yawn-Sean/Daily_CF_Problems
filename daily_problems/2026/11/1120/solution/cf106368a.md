**提示 1：** 取模运算只用做一次。

**提示 2：** 其实取模前也只有几种可能。

首先，取模操作只用做最后一次就好了。如果你做了几次取模操作，只需保留最后一次，结果不变。

因此有如下情况：

- 完全不用取模操作。

- 用一次取模操作：取模后的数只有两种可能，要么是一开始的 $a\bmod M$ ，要么是 $a\times M\bmod M$ ，不存在其它可能。

因此枚举上述三种情况。接下来只有加减法和乘法，就可以放心使用 BFS 求解最短路了，因为这三个操作在关于 $n$ 的取模里很好维护。

时间复杂度为 $\mathcal{O}(n)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    a, m, b, n = MII()
    
    sz = n + 5
    ans = ''
    
    def bfs(x, base):
        global sz, ans
        
        pre = [-2] * n
        pre_op = [0] * n
        
        pre[x] = -1
        que = [x]
        
        for u in que:
            nu = (u + m) % n
            if pre[nu] == -2:
                pre[nu] = u
                pre_op[nu] = 0
                que.append(nu)
            
            nu = (u - m) % n
            if pre[nu] == -2:
                pre[nu] = u
                pre_op[nu] = 1
                que.append(nu)
            
            nu = u * m % n
            if pre[nu] == -2:
                pre[nu] = u
                pre_op[nu] = 2
                que.append(nu)
        
        if pre[b] != -2:
            cur = b
            ops = []
    
            while cur != x:
                ops.append(pre_op[cur])
                cur = pre[cur]
            
            ops.reverse()
            res = base + ''.join("+-*"[x] for x in ops)
            
            if len(res) < sz:
                sz = len(res)
                ans = res
    
    bfs(a % n, '')
    bfs(a % m % n, '%')
    bfs(0, '*%')
    
    if sz <= n:
        print(sz)
        print(ans)
    else:
        print(-1)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int a, m, b, n;
	cin >> a >> m >> b >> n;

	int sz = n + 5;
	string ans;

	auto bfs = [&] (int x, string base) -> void {
		vector<int> pre(n, -2), pre_op(n);
		pre[x] = -1;
		
		vector<int> que = {x};

		for (int i = 0; i < que.size(); i ++) {
			int u = que[i], nu;

			nu = (u + m) % n;
			if (pre[nu] == -2) {
				pre[nu] = u;
				pre_op[nu] = 0;
				que.emplace_back(nu);
			}

			nu = ((u - m) % n + n) % n;
			if (pre[nu] == -2) {
				pre[nu] = u;
				pre_op[nu] = 1;
				que.emplace_back(nu);
			}

			nu = 1ll * u * m % n;
			if (pre[nu] == -2) {
				pre[nu] = u;
				pre_op[nu] = 2;
				que.emplace_back(nu);
			}
		}

		if (pre[b] != -2) {
			int cur = b;
			string ops;

			while (cur != x) {
				ops += "+-*"[pre_op[cur]];
				cur = pre[cur];
			}

			reverse(ops.begin(), ops.end());
			ops = base + ops;

			if (ops.size() < sz) {
				sz = ops.size();
				ans = ops;
			}
		}
	};

	bfs((a % n + n) % n, "");
	bfs((a % m + m) % m % n, "%");
	bfs(0, "*%");

	if (sz <= n) cout << sz << '\n' << ans << '\n';
	else cout << -1 << '\n';

	return 0;
}
```
