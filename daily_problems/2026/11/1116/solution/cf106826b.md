**提示 1：** 先拆分答案。再 DP 。

答案可以换个顺序算——可以算每个结点作为根时的合法排列数量，加总。这等价于每个排列合法的根的数量的和。

接下来就是典型的求每个点作为根的答案了，应该使用换根 DP 。

对于一个点，如果有 $1/2$ 个子树，答案就是子树的方案数的成绩再乘 $2$ （这是选择子树的左右），如果只有 $0$ 个子树，则方案数为 $1$ ，否则方案数为 $0$ 。

每个点再维护下其父结点对应的子树的方案数，接下来只需暴力枚举其邻接结点进行状态转移即可，因为每个点的度数都不大。

时间复杂度为 $\mathcal{O}(n)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    t = II()
    outs = []
    mod = 998244353
    
    for _ in range(t):
        n = II()
        path = [[] for _ in range(n)]
        
        for _ in range(n - 1):
            u, v = GMI()
            path[u].append(v)
            path[v].append(u)
        
        parent = [-1] * n
        que = [0]
        
        for u in que:
            for v in path[u]:
                if parent[u] != v:
                    parent[v] = u
                    que.append(v)
        
        dp = [1] * n
        
        for u in reversed(que):
            tmp = []
            
            for v in path[u]:
                if parent[v] == u:
                    tmp.append(v)
            
            if len(tmp) == 3:
                dp[u] = 0
                continue
            
            elif len(tmp):
                dp[u] = 2
                for v in tmp:
                    dp[u] = dp[u] * dp[v] % mod
        
        rev_dp = [1] * n
        
        for u in que:
            if u == 0:
                if len(path[u]) == 1:
                    rev_dp[path[u][0]] = rev_dp[u]
                
                else:
                    for x in path[0]:
                        rev_dp[x] = 2
                        for y in path[0]:
                            if y != x:
                                rev_dp[x] = rev_dp[x] * dp[y] % mod
            else:
                sons = []
                for v in path[u]:
                    if parent[v] == u:
                        sons.append(v)
                
                for x in sons:
                    rev_dp[x] = 2 * rev_dp[u] % mod
                    for y in sons:
                        if y != x:
                            rev_dp[x] = rev_dp[x] * dp[y] % mod
        
        ans = 0
        for i in range(n):
            if len(path[i]) <= 2:
                res = 2 * rev_dp[i] % mod
                for j in path[i]:
                    if parent[j] == i:
                        res = res * dp[j] % mod
                ans = (ans + res) % mod
        
        outs.append(ans)
    
    print('\n'.join(map(str, outs)))
```

C++ 做法如下——

```cpp []
using mint = atcoder::static_modint<998244353>;

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int t;
	cin >> t;

	while (t --) {
		int n;
		cin >> n;

		vector<vector<int>> path(n);
		for (int i = 0; i < n - 1; i ++) {
			int u, v;
			cin >> u >> v;
			u --, v --;
			path[u].emplace_back(v);
			path[v].emplace_back(u);
		}

		vector<mint> dp(n), rev_dp(n), parent(n, -1);

		auto dfs1 = [&] (auto &self, int u, int p) -> void {
			dp[u] = 1;
			parent[u] = p;

			vector<int> tmp;
			for (auto &v: path[u]) {
				if (v != p) {
					tmp.emplace_back(v);
				}
			}

			dp[u] = 2;
			for (auto &v: tmp) {
				self(self, v, u);
				dp[u] *= dp[v];
			}

			if (tmp.size() == 3) dp[u] = 0;
			else if (tmp.empty()) dp[u] = 1;
		};

		dfs1(dfs1, 0, -1);

		auto dfs2 = [&] (auto &self, int u, int p) -> void {
			if (u == 0) {
				if (path[u].size() == 1) {
					for (auto &v: path[u]) {
						rev_dp[v] = rev_dp[u];
						self(self, v, u);
					}
				}
				else {
					for (auto &x: path[u]) {
						rev_dp[x] = 2;
						for (auto &y: path[u]) {
							if (y != x) {
								rev_dp[x] *= dp[y];
							}
						}
						self(self, x, u);
					}
				}
			}
			else {
				vector<int> tmp;
				for (auto &v: path[u]) {
					if (v != p) {
						tmp.emplace_back(v);
					}
				}

				for (auto &x: tmp) {
					rev_dp[x] = 2 * rev_dp[u];
					for (auto &y: tmp) {
						if (y != x) {
							rev_dp[x] *= dp[y];
						}
					}
					self(self, x, u);
				}
			}
		};
	
		rev_dp[0] = 1;
		dfs2(dfs2, 0, -1);

		mint ans = 0;

		for (int i = 0; i < n; i ++) {
			if (path[i].size() <= 2) {
				mint res = 2 * rev_dp[i];
				for (auto &j: path[i]) {
					if (parent[j] == i) {
						res *= dp[j];
					}
				}
				ans += res;
			}
		}

		cout << ans.val() << '\n';
	}

	return 0;
}
```
