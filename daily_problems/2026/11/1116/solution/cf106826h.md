先看提示 1，发现每次取值范围得减半。所以第一次查 $x=2$ ，就可以判定是奇数还是偶数了。

接下来，发现可以二分。可以用位于中间的奇数，大于它的查的结果和小于它的查的结果不一致，因此总能使得范围不断砍半。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    def query(x):
        print('?', x, flush=True)
        return II()
    
    def answer(x):
        print('!', x)
    
    start = 1 if query(2) == 1 else 2
    
    l = 0
    r = (10 ** 9 - start) // 2
    
    while l < r:
        mid = (l + r) // 2
        
        w = start + 2 * mid + 1
        if w % 2 == 0: w += 1
        
        if query(w) == start % 2: r = mid
        else: l = mid + 1
    
    answer(start + 2 * l)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	auto query = [&] (int x) -> int {
		cout << "? " << x << endl;
		int v; cin >> v;
		return v;
	};

	auto answer = [&] (int x) -> void {
		cout << "! " << x << endl;
	};

	int start = query(2) ? 1 : 2;
	int l = 0, r = ((int)1e9 - start) / 2;

	while (l < r) {
		int mid = (l + r) / 2;
		int w = start + 2 * mid + 1;
		if (w % 2 == 0) w ++;

		if (query(w) == start % 2) r = mid;
		else l = mid + 1;
	}

	answer(start + 2 * l);

	return 0;
}
```
