**提示 1：** 给定一条路径，其期望结果如何？

**提示 2：** 避免精度损失。

给定一条路径，其期望踩雷数等于每个点是雷的概率的和。

我们要最小化这个和，你就需要维护到某个位置的最小概率和。这个可以用 DP 做。但概率本身是分数，所以你可以乘以分母的最小公倍数，这样用整数处理就可以无精度误差地求解 DP 了。

最后结果取模再乘以公分母的逆元即可。

时间复杂度为 $\mathcal{O}(nm)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main(): 
    W = 232792560
    
    m, n, k = MII()
    sx, sy, ex, ey = GMI()
    grid = [[0] * m for _ in range(n)]
    
    for _ in range(k):
        x1, y1, x2, y2 = MII()
        w = W // ((x2 - x1 + 1) * (y2 - y1 + 1))
        for i in range(x1 - 1, x2):
            for j in range(y1 - 1, y2):
                grid[i][j] += w
    
    for i in range(sx, ex + 1):
        for j in range(sy, ey + 1):
            if i == sx and j == sy: continue
            res = inf
            if i > sx: res = fmin(res, grid[i - 1][j])
            if j > sy: res = fmin(res, grid[i][j - 1])
            grid[i][j] += res
    
    mod = 10 ** 9 + 7
    print(grid[ex][ey] * pow(W, -1, mod) % mod)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int m, n, k;
	cin >> m >> n >> k;

	int sx, sy, ex, ey;
	cin >> sx >> sy >> ex >> ey;

	int W = 232792560;
	vector<vector<long long>> grid(n + 1, vector<long long>(m + 1, 0));

	while (k --) {
		int x1, y1, x2, y2;
		cin >> x1 >> y1 >> x2 >> y2;

		int w = W / (x2 - x1 + 1) / (y2 - y1 + 1);

		for (int x = x1; x <= x2; x ++) {
			for (int y = y1; y <= y2; y ++) {
				grid[x][y] += w;
			}
		}
	}

	for (int x = sx; x <= ex; x ++) {
		for (int y = sy; y <= ey; y ++) {
			if (x == sx && y == sy) continue;
			long long res = 1e18;
			if (x > sx) res = min(res, grid[x - 1][y]);
			if (y > sy) res = min(res, grid[x][y - 1]);
			grid[x][y] += res;
		}
	}

	int mod = 1e9 + 7;
	cout << grid[ex][ey] % mod * quickPow(W, mod - 2, mod) % mod;

	return 0;
}
```
