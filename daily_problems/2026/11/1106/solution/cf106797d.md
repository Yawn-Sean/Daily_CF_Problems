**提示 1：** 题目意味着任何两个序列的最长公共前缀长度。

**提示 2：** 从长往短考虑，则一堆可以合并。

题目相当于给定了一些无限长度的 01 串和部分串的最长公共前缀的长度，问包含 1 的最小数量。

从长往短考虑最长公共前缀的长度。

假设考虑到 $t$ ，此时意味着某些串的第 $t$ 位不一样而前面都一样。根据这个关系建边，形成的图只能是二分图（因为第 $t$ 位只有 0/1 两种选择）。

我们应该选择二分图中每个连通部分的最小的那个染色设定为 $1$ ，也就得到了这一位的 $1$ 的最小数量。

而连通块内部，前 $t-1$ 项应当完全一致，因此应该将连通块内的点进行合并，之后连边时，该点代表了一个点的集合，也就用 size 代替上面的数量统计。

时间复杂度为 $\mathcal{O}(n+m\log m)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main(): 
    n, m = MII()
    us = []
    vs = []
    ts = []
    
    for _ in range(m):
        u, v, t = MII()
        us.append(u - 1)
        vs.append(v - 1)
        ts.append(t)
    
    uf = UnionFind(n)
    
    tmp_path = [[] for _ in range(n)]
    cols = [-1] * n
    
    st_range = sorted(range(m), key=lambda x: -ts[x])
    l = 0
    
    ans = 0
    
    def solve(l, r):
        global ans
        tmp = []
        
        for i in range(l, r):
            u = us[st_range[i]]
            v = vs[st_range[i]]
            u = uf.find(u)
            v = uf.find(v)
            
            tmp_path[u].append(v)
            tmp_path[v].append(u)
            
            tmp.append(u)
            tmp.append(v)
        
        for u in tmp:
            if cols[u] == -1:
                cols[u] = 0
                que = [u]
                
                c0 = 0
                c1 = 0
    
                for u in que:
                    if cols[u] == 0: c0 += uf.getSize(u)
                    else: c1 += uf.getSize(u)
                    
                    for v in tmp_path[u]:
                        if cols[v] == -1:
                            cols[v] = cols[u] ^ 1
                            que.append(v)
                        elif cols[v] == cols[u]:
                            exit(print('impossible'))
                
                ans += fmin(c0, c1)
        
        for i in range(l, r):
            u = us[st_range[i]]
            v = vs[st_range[i]]
            uf.merge(u, v)
        
        for u in tmp:
            cols[u] = -1
            tmp_path[u].clear()
    
    for i in range(m):
        if ts[st_range[i]] != ts[st_range[l]]:
            solve(l, i)
            l = i
    
    solve(l, m)
    print('possible')
    print(ans)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(nullptr);

	int n, m;
	cin >> n >> m;

	vector<array<int, 3>> edges(m);
	for (auto &[u, v, w]: edges) {
		cin >> u >> v >> w;
		u --, v --;
	}

	sort(edges.begin(), edges.end(), [&] (array<int, 3> &x, array<int, 3> &y) {
		return x[2] > y[2];
	});

	int l = 0, ans = 0;
	atcoder::dsu uf(n);
	vector<vector<int>> tmp_path(n);
	vector<int> cols(n, -1);

	auto solve = [&] (int l, int r) -> void {
		vector<int> tmp;

		for (int i = l; i < r; i ++) {
			auto [u, v, t] = edges[i];
			u = uf.leader(u), v = uf.leader(v);
			tmp.emplace_back(u), tmp.emplace_back(v);
			tmp_path[u].emplace_back(v), tmp_path[v].emplace_back(u);
		}

		for (auto &u: tmp) {
			if (cols[u] == -1) {
				int c0 = 0, c1 = 0;
				vector<int> stk = {u};

				cols[u] = 0;

				while (!stk.empty()) {
					auto u = stk.back();
					stk.pop_back();

					if (cols[u] == 0) c0 += uf.size(u);
					else c1 += uf.size(u);

					for (auto &v: tmp_path[u]) {
						if (cols[v] == -1) {
							cols[v] = cols[u] ^ 1;
							stk.emplace_back(v);
						}
						else if (cols[u] == cols[v]) {
							cout << "impossible\n";
							exit(0);
						}
					}
				}

				ans += min(c0, c1);
			}
		}

		for (int i = l; i < r; i ++) {
			auto [u, v, t] = edges[i];
			uf.merge(u, v);
		}

		for (auto &u: tmp) cols[u] = -1, tmp_path[u].clear();
	};

	for (int i = 0; i < m; i ++) {
		if (edges[i][2] != edges[l][2]) {
			solve(l, i);
			l = i;
		}
	}

	solve(l, m);

	cout << "possible\n";
	cout << ans << '\n';

	return 0;
}
```
