**提示 1：** DP ，想想状态和状态转移。

很明显，我们令前 $i$ 天的旅行计划的最小成本是 $dp[i]$ 。接下来会发生什么转移呢？

- 直接购入新的一天的单天票，即 $dp[i]+f\to dp[i+1]$ 。

- 选择套票。对于起点 $i+1$ ，我们使用套票最多能到第几天呢？只需判两件事——之间的总天数是否不超过 $p$ ，之间的旅行天数是否不超过 $d$ 。可以二分找对应位置，也可以选择用指针维护，不断右移即可。

每个状态有 $\mathcal{O}(k)$ 转移。

时间复杂度为 $\mathcal{O}(nk)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n, k = MII()
    
    ts = []
    fs = []
    
    for _ in range(n):
        t, f = MII()
        ts.append(t)
        fs.append(f)
    
    ps = []
    ds = []
    cs = []
    
    for _ in range(k):
        p, d, c = MII()
        ps.append(p)
        ds.append(d)
        cs.append(c)
    
    dp = [inf] * (n + 1)
    dp[0] = 0
    
    pts = [0] * k
    
    for i in range(n):
        dp[i + 1] = dp[i] + fs[i]
        
        for j in range(k):
            while ts[i] - ts[pts[j]] + 1 > ps[j] or i - pts[j] + 1 > ds[j]:
                pts[j] += 1
            
            dp[i + 1] = fmin(dp[i + 1], dp[pts[j]] + cs[j])
    
    print(dp[n])
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n, k;
	cin >> n >> k;

	vector<pair<int, int>> days(n);
	for (auto &[t, f]: days) cin >> t >> f;

	vector<array<int, 3>> tickets(k);
	for (auto &[p, d, c]: tickets) cin >> p >> d >> c;

	vector<int> dp(n + 1, 1e9);
	dp[0] = 0;

	vector<int> pts(k, 0);

	for (int i = 0; i < n; i ++) {
		auto [t, f] = days[i];
		dp[i + 1] = dp[i] + f;

		for (int j = 0; j < k; j ++) {
			auto [p, d, c] = tickets[j];
			while (t - days[pts[j]].first + 1 > p || i - pts[j] + 1 > d) pts[j] ++;
			dp[i + 1] = min(dp[i + 1], dp[pts[j]] + c);
		}
	}

	cout << dp[n];

	return 0;
}
```
