**提示 1：** 两个问题都考虑太难了，想办法简化其一。

我们可以对于每一个 $x$ ，求能被 $x$ 整除的不同递增子序列的数量——只需遍历一个新元素时，更新以这个元素为结尾的递增子序列的个数（只需得到当前结尾元素比它小的序列的个数的和就行）。

而我们对于 $1\sim M$ 的元素全部统计一下上面的结果，就可以得到所有元素都能被 $x$ 整除的递增子序列的个数，也就是 GCD 是 $x$ 的倍数的递增子序列的个数。

那考虑 GCD 恰好是 $x$ 的递增子序列的个数，就等于 GCD 是 $x$ 的倍数的子序列的个数，减去恰好是 $2x,3x,\dots$ 的子序列的个数，所以从大到小遍历 $x$ ，按照上面的逻辑排除即可。

时间复杂度为 $\mathcal{O}(\sum d(a_i)\log n+M\log M)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    M = 10 ** 6 + 5
    pr = list(range(M))
    
    for i in range(2, M):
        if pr[i] == i:
            for j in range(i, M, i):
                pr[j] = i
    
    n = II()
    mod = 10 ** 9 + 7
    
    v1 = LII()
    v2 = LII()
    
    def factors(x):
        ans = [1]
        while x > 1:
            p = pr[x]
            c = 0
            while x % p == 0:
                x //= p
                c += 1
            
            for i in range(c * len(ans)):
                ans.append(ans[i] * p)
        
        return ans
    
    pos = [[] for _ in range(M)]
    
    for i in range(n):
        for x in factors(v2[i]):
            pos[x].append(i)
    
    del pr
    
    sorted_v1 = sorted(v1)
    v1 = [bisect.bisect_left(sorted_v1, x) for x in v1]
    
    del sorted_v1
    
    cnt = [0] * M
    tmp = [0] * n
    
    fen = FenwickTree(n)
    
    for i in range(M):
        for j in pos[i]:
            tmp[j] = (fen.rsum(0, v1[j] - 1) + 1) % mod
            cnt[i] = (cnt[i] + tmp[j]) % mod
            fen.add(v1[j], tmp[j])
        
        for j in pos[i]:
            fen.add(v1[j], -tmp[j])
    
    for i in range(M - 1, 0, -1):
        for j in range(2 * i, M, i):
            cnt[i] -= cnt[j]
            cnt[i] %= mod
    
    print(sum(i * cnt[i] % mod for i in range(M)) % mod)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int M = 1e6 + 5, mod = 1e9 + 7;

	vector<int> pr(M);
	iota(pr.begin(), pr.end(), 0);

	for (int i = 2; i < M; i ++) {
		if (pr[i] == i) {
			for (int j = i; j < M; j += i) {
				pr[j] = i;
			}
		}
	}

	auto factors = [&] (int x) -> vector<int> {
		vector<int> ans = {1};
		while (x > 1) {
			int p = pr[x], c = 0, l = ans.size();
			while (x % p == 0) x /= p, c ++;
			for (int i = 0; i < c * l; i ++) ans.emplace_back(ans[i] * p);
		}
		return ans;
	};

	int n;
	cin >> n;

	vector<int> v1(n), v2(n);
	for (auto &x: v1) cin >> x;
	for (auto &x: v2) cin >> x;

	vector<int> sorted_v1 = v1;
	sort(sorted_v1.begin(), sorted_v1.end());

	for (int i = 0; i < n; i ++)
		v1[i] = lower_bound(sorted_v1.begin(), sorted_v1.end(), v1[i]) - sorted_v1.begin();

	vector<vector<int>> pos(M);
	for (int i = 0; i < n; i ++) {
		for (auto &x: factors(v2[i])) {
			pos[x].emplace_back(i);
		}
	}

	atcoder::fenwick_tree<long long> fen(n);
	vector<int> tmp(n, 0), cnt(M, 0);

	for (int i = 0; i < M; i ++) {
		for (auto &p: pos[i]) {
			tmp[p] = (fen.sum(0, v1[p]) + 1) % mod;
			cnt[i] = (cnt[i] + tmp[p]) % mod;
			fen.add(v1[p], tmp[p]);
		}
		for (auto &p: pos[i]) {
			fen.add(v1[p], -tmp[p]);
		}
	}

	int ans = 0;

	for (int i = M - 1; i >= 1; i --) {
		for (int j = 2 * i; j < M; j += i) {
			cnt[i] += mod - cnt[j];
			cnt[i] %= mod;
		}
		ans = (ans + 1ll * i * cnt[i]) % mod;
	}

	cout << ans;

	return 0;
}
```
