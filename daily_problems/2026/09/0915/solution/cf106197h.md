**提示 1：** 找到一个最小的不可达的集合。怎么做？

我们尝试找一个不可达的集合。

首先，这个集合包含 $n$ 。

而任意一个不在这个集合的点，都得满足其只有至多一半的边指向这个集合。一旦不满足这个条件，这个点就得加入集合中。

我们可以不断执行这个操作，怎么执行呢？直接用反图，找到指向它的点，维护每个点指向集合的边数。此时，一旦边数超越了界限，这个点加入集合。

整个写法类似于拓扑排序。

这样，我们找到了不可达的点形成的最小集合。而一旦这个集合包含 $1$ ，也就推出了矛盾。

否则，我们只需去掉那些从可达位置指向不可达位置的边即可。合法性显然。

时间复杂度为 $\mathcal{O}(n+m)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    t = II()
    outs = []
    
    for _ in range(t):
        n, m = MII()
        rev_path = [[] for _ in range(n)]
        out_deg = [0] * n
    
        for _ in range(m):
            u, v = GMI()
            rev_path[v].append(u)
            out_deg[u] += 1
        
        vis = [0] * n
        cur_deg = [0] * n
        
        vis[n - 1] = 1
        
        stk = [n - 1]
        
        while stk:
            u = stk.pop()
            
            for v in rev_path[u]:
                cur_deg[v] += 1
                
                if 2 * cur_deg[v] > out_deg[v] and not vis[v]:
                    vis[v] = 1
                    stk.append(v)
        
        if vis[0]: outs.append('-1')
        else:
            edges = []
            
            for u in range(n):
                for v in rev_path[u]:
                    if not vis[v] and vis[u]:
                        edges.append(f'{v + 1} {u + 1}')
            
            outs.append(f'{len(edges)}')
            
            for e in edges: outs.append(e)
    
    print('\n'.join(outs))
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int t;
	cin >> t;

	while (t --) {
		int n, m;
		cin >> n >> m;

		vector<int> out_deg(n, 0);
		vector<vector<int>> rev_path(n);

		while (m --) {
			int u, v;
			cin >> u >> v;
			u --, v --;

			out_deg[u] ++;
			rev_path[v].emplace_back(u);
		}

		vector<int> vis(n, 0), cur_deg(n, 0);
		vis[n - 1] = 1;

		auto dfs = [&] (auto &self, int u) -> void {
			for (auto &v: rev_path[u]) {
				cur_deg[v] ++;
				if (cur_deg[v] * 2 > out_deg[v] && !vis[v]) {
					vis[v] = 1;
					self(self, v);
				}
			}
		};

		dfs(dfs, n - 1);

		if (vis[0]) cout << "-1\n";
		else {
			vector<pair<int, int>> edges;

			for (int u = 0; u < n; u ++) {
				for (auto &v: rev_path[u]) {
					if (!vis[v] && vis[u]) {
						edges.emplace_back(v, u);
					}
				}
			}

			cout << edges.size() << '\n';
			for (auto &[u, v]: edges) cout << u + 1 << ' ' << v + 1 << '\n';
		}
	}

	return 0;
}
```
