**提示 1：** 考虑 DP 状态。

考虑到第 $i$ 天为止，最后一个使用的活动是第 $j$ 个的情况下的最优分数 $dp[i][j]$ 。

则我们可以继续做这个活动，我们认为此时已经到了 $c\gt d_i$ 的时期，因此将 $dp[i][j]+k_j-x_j$ 转移到 $dp[i+1][j]$ ，这样可以统一处理后面 $k-x$ 的部分，比较方便

接下来我们考虑 $c\leq d_i$ 的部分。此时假设转移到 $dp[i+c][j]$ ，则 $c$ 应当枚举 $1\sim d_i$ 的任意值。同时转移前使用的活动必须不等于 $j'$ ，所以我们需要找到 $\max_{j'\neq j} dp[i][j']$ ，加上 $ck_i$ 得到转移，为此，只需维护 $i$ 那一层的最大值和次大值即可。

时间复杂度为 $\mathcal{O}(nz\max(d))$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n, z = MII()
    
    ks = [0]
    ds = [1]
    xs = [0]
    
    for _ in range(n):
        k, d, x = MII()
        ks.append(k)
        ds.append(d)
        xs.append(x)
    
    dp = [[0] * (n + 1) for _ in range(z + 1)]
    
    for i in range(z):
        ma1 = 0
        ma2 = 0
        
        for j in range(n + 1):
            dp[i + 1][j] = fmax(dp[i + 1][j], dp[i][j] + ks[j] - xs[j])
            
            if dp[i][j] > ma1: ma1, ma2 = dp[i][j], ma1
            elif dp[i][j] > ma2: ma2 = dp[i][j]
        
        for j in range(n + 1):
            nd = fmin(z - i, ds[j])
            
            for x in range(1, nd + 1):
                dp[i + x][j] = fmax(dp[i + x][j], (ma2 if dp[i][j] == ma1 else ma1) + x * ks[j])
    
    print(max(dp[z]))
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n, z;
	cin >> n >> z;

	vector<array<int, 3>> activities(n);
	for (auto &[k, d, x]: activities) cin >> k >> d >> x;

	activities.push_back({0, 1, 0});

	vector<vector<long long>> dp(z + 1, vector<long long>(n + 1, 0));
	
	for (int i = 0; i < z; i ++) {
		long long ma1 = 0, ma2 = 0;

		for (int j = 0; j <= n; j ++) {
			auto [k, d, x] = activities[j];
			dp[i + 1][j] = max(dp[i + 1][j], dp[i][j] + k - x);

			if (dp[i][j] > ma1) swap(ma1, ma2), ma1 = dp[i][j];
			else if (dp[i][j] > ma2) ma2 = dp[i][j];
		}

		for (int j = 0; j <= n; j ++) {
			auto [k, d, x] = activities[j];
			int nd = min(z - i, d);

			for (int v = 1; v <= nd; v ++)
				dp[i + v][j] = max(dp[i + v][j], (dp[i][j] == ma1 ? ma2 : ma1) + 1ll * v * k);
		}
	}

	cout << *max_element(dp[z].begin(), dp[z].end());

	return 0;
}
```
