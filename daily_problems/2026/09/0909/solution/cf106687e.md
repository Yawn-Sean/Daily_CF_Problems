**提示 1：** 最短路问题。如何比较两个长度？

还是一个最短路题——求的顺序依旧是从最短的路径出发，按照最短路的长度顺序依次推出别的各个其它的点。这主要是因为最短路更长的点能保证是由更短的点加一条边得到的。

所以考虑最短路。

有两种方式理解：

**方法 1：** 把最短路长度视为 $k$ 进制数。

这样，我们去掉前导 $0$ 后，只需先比较位数，再比较字典序就可以判定长度了。

而前导 $0$ 怎么去掉呢？直接从 $1$ 结点出发，关于所有 $0$ 边权的边进行 BFS 就行。

接下来，所有走的边都会使得最短路的路径长度 $+1$ ，也就是数值更大了。所以接下来求最短路，首先要保证走的边数最少，在这个基础上字典序最小。

因此可以层序遍历——依次考虑最少经过 $1,2,3,\dots$ 层的那些点。

我们可以利用前一层的字典序的比较关系和新的边的比较关系，得到新的一层的字典序比较关系，注意这里可能相等，需要赋一样的值。

**方法 2：** 魔改 Dijkstra 。

直接存最短路长度是不现实的，因为太长了。

那怎么比较呢？如果两个结点的前驱结点一个比另一个最短路严格小，那么这两个结点也满足对应的大小关系。否则，就取决于其和前驱节点连的边。

基于这个认识，我们可以用 $(前驱节点最短路的长度排名,新边边权)$ 表示每个点的最短路长度标记，用这件事来记录距离，跑最短路。注意排名是如何修改的——只有严格更大了，这个数值才会 $+1$ 。

时间复杂度为 $\mathcal{O}(m+n\log m)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    t = II()
    outs = []
    
    mod = 998244353
    
    for _ in range(t):
        n, m, k = MII()
        path = [[] for _ in range(n)]
        
        for _ in range(m):
            u, v, w = MII()
            u -= 1
            v -= 1
            path[u].append(w * n + v)
            path[v].append(w * n + u)
        
        def f(prev_dis, edge, u):
            return (prev_dis * k + edge) * n + u
        
        dis = [(n, k)] * n
        dis[0] = (0, 0)
        
        dis_idx = [n] * n
        dis_idx[0] = 0
    
        ans = [-1] * n
        ans[0] = 0
        
        pq = [f(0, 0, 0)]
        
        cur_prev_dis = 0
        cur_edge = 0
        cur_idx = 0
        
        while pq:
            msk = heappop(pq)
            
            msk, u = divmod(msk, n)
            prev_dis, edge = divmod(msk, k)
            
            if dis[u] == (prev_dis, edge):
                if (prev_dis, edge) > (cur_prev_dis, cur_edge):
                    cur_prev_dis = prev_dis
                    cur_edge = edge
                    cur_idx += 1
                dis_idx[u] = cur_idx
                
                for msk in path[u]:
                    d, v = divmod(msk, n)
                    if dis[v] > (dis_idx[u], d):
                        dis[v] = (dis_idx[u], d)
                        ans[v] = (ans[u] * k + d) % mod
                        heappush(pq, f(dis_idx[u], d, v))
        
        outs.append(' '.join(map(str, ans)))
    
    print('\n'.join(outs))
```

C++ 做法如下——

```cpp []
typedef pair<pair<int, int>, int> node;

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int t, mod = 998244353;
	cin >> t;

	while (t --) {
		int n, m, k;
		cin >> n >> m >> k;

		vector<vector<pair<int, int>>> path(n);

		while (m --) {
			int u, v, w;
			cin >> u >> v >> w;
			u --, v --;
			path[u].emplace_back(v, w);
			path[v].emplace_back(u, w);
		}

		vector<pair<int, int>> dis(n, {n, k});
		vector<int> dis_idx(n, n), ans(n, -1);

		dis[0] = {0, 0};
		dis_idx[0] = 0;
		ans[0] = 0;

		priority_queue<node, vector<node>, greater<node>> pq;
		pq.push({dis[0], 0});

		pair<int, int> cur_node = {0, 0};
		int cur_idx = 0;

		while (!pq.empty()) {
			auto cur = pq.top(); pq.pop();
			auto [prev_idx, edge] = cur.first;
			int u = cur.second;

			if (dis[u] == cur.first) {
				if (cur.first > cur_node) {
					cur_node = cur.first;
					cur_idx ++;
				}
				dis_idx[u] = cur_idx;

				for (auto &[v, w]: path[u]) {
					pair<int, int> ncur = {dis_idx[u], w};
					if (dis[v] > ncur) {
						dis[v] = ncur;
						ans[v] = (1ll * ans[u] * k + w) % mod;
						pq.push({dis[v], v});
					}
				}
			}
		}

		for (int i = 0; i < n; i ++) cout << ans[i] << " \n"[i == n - 1];
	}

	return 0;
}
```
