**提示 1：** 考虑必须填入的是哪些数，填入他们，剩余部分再任意填。

首先，我们考虑每一列。

对于那些恰好只缺一个数的列，都可能作为产生答案的备选。统计这些元素的出现频率。

因为每个这样的数字都在不同的列，所以互不影响。

因此，分别给这些数分配一个位置，对于剩下的数，再进行任意排列即可。

时间复杂度为 $\mathcal{O}(n^2)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n = II()
    grid = [LII() for _ in range(n - 1)]
    
    vis = [0] * (n + 1)
    cnt = [0] * (n + 1)
    
    for i in range(n):
        for j in range(n - 1):
            vis[grid[j][i]] = 1
    
        c = 0
        target = 0
        
        for j in range(1, n + 1):
            if vis[j] == 0:
                c += 1
                target = j
        
        if c == 1: cnt[target] += 1
        
        for j in range(n - 1):
            vis[grid[j][i]] = 0
    
    ans0 = 0
    ans1 = 1
    
    resid = n
    mod = 10 ** 9 + 7
    
    for i in range(n + 1):
        if cnt[i]:
            ans0 += 1
            ans1 = ans1 * cnt[i] % mod
            resid -= 1
    
    for i in range(1, resid + 1):
        ans1 = ans1 * i % mod
    
    print(ans0, ans1)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n;
	cin >> n;

	vector<vector<int>> grid(n - 1, vector<int>(n));
	for (auto &x: grid) for (auto &y: x) cin >> y;

	vector<int> vis(n + 1, 0), cnt(n + 1, 0);

	for (int i = 0; i < n; i ++) {
		for (int j = 0; j < n - 1; j ++) {
			vis[grid[j][i]] = 1;
		}

		int c = 0, target = 0;
		for (int i = 1; i <= n; i ++) {
			if (!vis[i]) {
				c ++;
				target = i;
			}
		}

		if (c == 1) cnt[target] ++;

		for (int j = 0; j < n - 1; j ++) {
			vis[grid[j][i]] = 0;
		}
	}

	int ans0 = 0, ans1 = 1, mod = 1e9 + 7;

	for (auto &x: cnt) {
		if (x) {
			ans0 ++;
			ans1 = 1ll * ans1 * x % mod;
		}
	}

	for (int i = 1; i <= n - ans0; i ++) {
		ans1 = 1ll * ans1 * i % mod;
	}

	cout << ans0 << ' ' << ans1 << '\n';

	return 0;
}
```
