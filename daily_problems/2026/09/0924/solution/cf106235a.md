**提示 1：** 哪些人会必胜？

我们用总组数 $\binom{n}{k}$ 减去不合法的组数即可。

什么情况下会有个人必胜呢？就是这个人的两个参数比其余的任何人都高。

所以我们枚举这个人，统计两项都严格劣于这个人的人数 $c$ ，减去分别的 $\binom{c}{k}$ 即可。

时间复杂度为 $\mathcal{O}(n^2)$ ，如果用二维数点可以进一步优化到 $\mathcal{O}(n\log n)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n, k = MII()
    p = LII()
    s = LII()
    
    mod = 998244353
    f = Factorial(n, mod)
    
    ans = f.combi(n, k)
    
    for i in range(n):
        c = 0
        for j in range(n):
            if p[j] < p[i] and s[j] < s[i]:
                c += 1
        
        ans -= f.combi(c, k - 1)
        ans %= mod
    
    print(ans)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n, k, mod = 998244353;
	cin >> n >> k;

	vector<int> p(n), s(n);
	for (auto &x: p) cin >> x;
	for (auto &x: s) cin >> x;

	vector<int> f(n + 1), g(n + 1);
	f[0] = 1;
	for (int i = 1; i <= n; i ++) f[i] = 1ll * f[i - 1] * i % mod;
	g[n] = quickPow(f[n], mod - 2, mod);
	for (int i = n; i >= 1; i --) g[i - 1] = 1ll * g[i] * i % mod;

	auto comb = [&] (int x, int y) -> int {
		if (y < 0 || y > x) return 0;
		return 1ll * f[x] * g[y] % mod * g[x - y] % mod;
	};

	int ans = comb(n, k);

	for (int i = 0; i < n; i ++) {
		int c = 0;
		for (int j = 0; j < n; j ++) {
			if (p[j] < p[i] && s[j] < s[i]) {
				c ++;
			}
		}
		ans += mod - comb(c, k - 1);
		ans %= mod;
	}

	cout << ans;

	return 0;
}
```
