**提示 1：** 考虑所有的间隔，你枚举中转站，应该怎么做呢？

我们只需考虑 Alice 和 Bob 那些有重叠的间隔。为此只需使用一个双指针就能找到这些间隔对。

接下来问题就是， Alice 在 $[l_1,r_1]$ 间要从 $s_1$ 到 $t_1$ ， Bob 在 $[l_2,r_2]$ 间要从 $s_2$ 到 $t_2$ ，问中间最多相遇多久。

我们枚举中转站。只有一个相遇的位置，不然的话可以在第一个位置相遇足够多时间之后再走，效果不劣。

所以得先确认 $s_1$ 到 $i$ 和 $i$ 到 $t_1$ 最短路，这件事儿可以预处理。然后我们就可以得到最早相遇时间和最迟分离时间，两者之差就是最大相遇时间。

注意可能不能相遇，此时这一段答案为 $0$ 就行。

把各段加总即可。

时间复杂度为 $\mathcal{O}(n^3+n(a+b))$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    t = II()
    outs = []
    
    for _ in range(t):
        n, m = MII()
        
        dis = [[inf] * n for _ in range(n)]
        for i in range(n):
            dis[i][i] = 0
        
        for _ in range(m):
            u, v, w = MII()
            u -= 1
            v -= 1
            dis[u][v] = fmin(dis[u][v], w)
            dis[v][u] = fmin(dis[v][u], w)
        
        for i in range(n):
            for j in range(n):
                for k in range(n):
                    dis[j][k] = fmin(dis[j][k], dis[j][i] + dis[i][k])
        
        a, b = MII()
        
        alice = [tuple(MII()) for _ in range(a)]
        bob = [tuple(MII()) for _ in range(b)]
        
        ans = 0
        pta, ptb = 0, 0
        
        while pta + 1 < a and ptb + 1 < b:
            l1, r1, s1, t1 = alice[pta][1], alice[pta + 1][0], alice[pta][2], alice[pta + 1][2]
            l2, r2, s2, t2 = bob[ptb][1], bob[ptb + 1][0], bob[ptb][2], bob[ptb + 1][2]
            s1 -= 1
            t1 -= 1
            s2 -= 1
            t2 -= 1
            
            res = 0
            
            for i in range(n):
                l = fmax(l1 + dis[s1][i], l2 + dis[s2][i])
                r = fmin(r1 - dis[i][t1], r2 - dis[i][t2])
                res = fmax(res, r - l)
            
            ans += res
            
            if r1 < r2: pta += 1
            else: ptb += 1
        
        outs.append(ans)
    
    print('\n'.join(map(str, outs)))
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int t;
	cin >> t;

	long long inf = 1e18;

	while (t --) {
		int n, m;
		cin >> n >> m;

		vector<vector<long long>> dis(n, vector<long long>(n, inf));

		for (int i = 0; i < n; i ++) dis[i][i] = 0;

		while (m --) {
			int u, v;
			long long w;
			cin >> u >> v >> w;
			u --, v --;

			dis[u][v] = min(dis[u][v], w);
			dis[v][u] = min(dis[v][u], w);
		}

		for (int i = 0; i < n; i ++) {
			for (int j = 0; j < n; j ++) {
				for (int k = 0; k < n; k ++) {
					dis[j][k] = min(dis[j][k], dis[j][i] + dis[i][k]);
				}
			}
		}

		int a, b;
		cin >> a >> b;

		vector<array<long long, 3>> alice(a), bob(b);
		for (auto &[s, e, l]: alice) cin >> s >> e >> l, l --;
		for (auto &[s, e, l]: bob) cin >> s >> e >> l, l --;

		long long ans = 0;
		int pta = 0, ptb = 0;

		while (pta + 1 < a && ptb + 1 < b) {
			long long l1 = alice[pta][1], r1 = alice[pta + 1][0];
			int s1 = alice[pta][2], t1 = alice[pta + 1][2];

			long long l2 = bob[ptb][1], r2 = bob[ptb + 1][0];
			int s2 = bob[ptb][2], t2 = bob[ptb + 1][2];

			long long res = 0;
			for (int i = 0; i < n; i ++) {
				long long l = max(l1 + dis[s1][i], l2 + dis[s2][i]);
				long long r = min(r1 - dis[i][t1], r2 - dis[i][t2]);
				res = max(res, r - l);
			}

			ans += res;

			if (r1 < r2) pta ++;
			else ptb ++;
		}

		cout << ans << '\n';
	}

	return 0;
}
```
