**提示 1：** 首先，每个元素可以直接转化为 $\mathrm{gcd}(x, nums[i])$ ，此时不同的元素就不多了。

**提示 2：** 转化后，相同的元素没必要用太多次。

首先，一个元素除去 $\mathrm{gcd}(x, nums[i])$ 部分的因子没办法再对乘积是否被 $x$ 整除有贡献，所以直接用这个新元素替代。这样整个数组中不同的元素就不超过 $x$ 的因子个数。

接下来从小到大考虑每个数组中的转化后的元素。

此时考虑因子 $V$ 。假设考虑了所有小于 $V$ 的因子时，前选取元素集和 $x$ 的最大公约数为 $y$ 的最小子集大小为 $dp[y]$ ，则此时尝试不断往集合中加入一个个的元素 $V$ ，则 $y$ 状态可以转移为 $\mathrm{gcd}(yV,x)$ ，于是可以用 $dp[y]+1$ 更新 $dp[\mathrm{gcd}(yV,x)]$ 的数值。

但如果这样一直做，总复杂度是 $\mathcal{O}(nd(x)\log x)$ 的，那怎么优化呢？

事实上， $V$ 最多加入集合 $\mathcal{O}(\log x)$ 次，因为对于 $x$ 的每一个质因子，次数不超过 $\mathcal{O}(\log x)$ 量级，而如果要用 $V$ 来凑这些次数，最多也只需要 $\mathcal{O}(\log x)$ 这么多个。

综上，我们只需使用 $V$ 直到 DP 数组不更新即可。

另外，这里求最大公约数的部分可能带来额外的成本，你可以预处理，这样会更快一点，也就是任意两个位置的因子乘起来，和 $x$ 求出的最大公约数在哪个位置。

时间复杂度为 $\mathcal{O}(n\log x+d(x)^2\log x)$ ，其中 $d(x)$ 表示 $x$ 的因子个数。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    n, x = MII()
    
    factors = []
    
    for i in range(1, 100000):
        if x // i < i: break
        if x % i == 0:
            factors.append(i)
            if x // i != i:
                factors.append(x // i)
    
    factors.sort()
    
    k = len(factors)
    d = {v: i for i, v in enumerate(factors)}
    
    cnt = [0] * k
    
    for v in MII():
        cnt[d[math.gcd(v, x)]] += 1
    
    gcds = [[0] * k for _ in range(k)]
    
    for i in range(k):
        for j in range(k):
            gcds[i][j] = math.gcd(factors[i], factors[j])
    
    dp = [inf] * k
    dp[0] = 0
    
    for i in range(k):
        for _ in range(cnt[i]):
            ndp = dp[:]
            flg = False
            
            for j in range(k):
                nj = d[factors[j] * gcds[d[x // factors[j]]][i]]
                if ndp[nj] > dp[j] + 1:
                    ndp[nj] = dp[j] + 1
                    flg = True
            
            if not flg:
                break
            
            dp = ndp
    
    print(dp[-1] if dp[-1] < inf else -1)
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int n, x;
	cin >> n >> x;

	vector<int> factors;

	for (int i = 1; i <= 100000; i ++) {
		if (x / i < i) break;
		if (x % i == 0) {
			factors.emplace_back(i);
			if (x / i != i) factors.emplace_back(x / i);
		}
	}

	int k = factors.size();
	sort(factors.begin(), factors.end());

	auto pos = [&] (int x) -> int {
		return lower_bound(factors.begin(), factors.end(), x) - factors.begin();
	};

	vector<int> cnt(k, 0);

	for (int i = 0; i < n; i ++) {
		long long v; cin >> v;
		cnt[pos(gcd(v, x))] ++;
	}

	vector<vector<int>> mapping(k, vector<int>(k));

	for (int i = 0; i < k; i ++) {
		for (int j = 0; j < k; j ++) {
			mapping[i][j] = pos(factors[i] * gcd(x / factors[i], factors[j]));
		}
	}

	vector<int> dp(k, n + 5);
	dp[0] = 0;

	for (int i = 0; i < k; i ++) {
		while (cnt[i] --) {
			vector<int> ndp = dp;
			bool flg = false;

			for (int j = 0; j < k; j ++) {
				int nj = mapping[i][j];
				if (ndp[nj] > dp[j] + 1) {
					ndp[nj] = dp[j] + 1;
					flg = true;
				}
			}

			if (!flg) break;

			dp.swap(ndp);
		}
	}

	cout << (dp[k - 1] <= n ? dp[k - 1] : -1);

	return 0;
}
```
