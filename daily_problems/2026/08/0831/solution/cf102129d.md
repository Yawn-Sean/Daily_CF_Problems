**提示 1：** 用 $a_1,a_2,\dots,a_n$ 表示 $F_n$ 和 $F_{n-b_i}$ 。接下来就是解方程了。

一个序列可以由前 $k$ 项完全确定。而如果我们求出了 $F_x=v_{1,x}F_1+v_{2,x}F_2+\dots+v_{k,x}F_k$ ，则 $F_n=\sum\limits_{i=1}^kc_iF_{n-b_i}$ 的两端都可以表示为 $F_1,F_2,\dots,F_k$ 的线性组合。这样，要使得等式恒成立，我们就只需分别对比它们的系数，必须严格相等。

这样我们可以列出 $k$ 个方程，而我们恰好有 $c_1,c_2,\dots,c_k$ 这些变量，所以刚好足够解出。（这里矩阵可逆性说明相对没那么容易，可能得转成多项式理解）

这样，我们的问题就只是，将数组中的每个元素用前 $k$ 项表示出来，也就是求解 $F_{x+1}=v_{1,x+1}F_1+v_{2,x+1}F_2+\dots+v_{k,x+1}F_k$ 中的各个系数。

这个问题，可以利用类似快速幂的方法求解：

如果 $F_{x+1}=v_{1,x+1}F_1+v_{2,x+1}F_2+\dots+v_{k,x+1}F_k, F_{y+1}=v_{1,y+1}F_1+v_{2,y+1}F_2+\dots+v_{k,y+1}F_k$ ，也就是找到了前 $k$ 项和第 $x$ 项与第 $y$ 项的关系。

则类似地，我们将这个序列平移，即将第 $j+1,j+2,\dots,j+k$ 项视为前 $k$ 项，则第 $j+x$ 项和第 $j+y$ 项的表示系数也是一样的。

因此，有：

$F_{x+y+1}=v_{1,x+1}F_{y+1}+v_{2,x+1}F_{y+2}+\dots+v_{k,x+1}F_{y+k}$

其中 $F_{y+i}=v_{1,y+1}F_{i}+v_{2,y+1}F_{i+1}+\dots+v_{k,y+1}F_{i+k}$ （第二个式子平移 $i-1$ 位）。

这样，可以得到： $F_{x+y+1}=\sum\left(\sum\limits_{i+j+V=x+y}v_{i,x+1}v_{j,y+1}\right)F_{V+1}$ 。

所以是个卷积的问题，每一项的系数都有 $\mathcal{O}(n)$ 项数，因此所有系数的求解加起来是 $\mathcal{O}(n^2)$ 的。而这里 $V$ 的数值并不会过大，因为 $x+y\leq 2k$ ，所以最后可以利用前 $2k$ 项的线性表示的预处理，再将这里的 $F_{V+1}$ 展开。

而这也就说明，只要我们求出 $F_{n+1}$ ， $F_{2n+1}$ 是很容易的。

同时， $F_{2n+2}$ 可以用 $F_{2n+1}$ 的结果，即平移一位，用 $F_2,\dots,F_{k+1}$ 表示 $F_{2n+2}$ ： $F_{2n+2}=v_{1,2n+1}F_2+v_{2,2n+1}F_3+\dots+v_{k,2n+1}F_{k+1}$ 。再将这里的 $F_{k+1}$ 展开为 $F_1,\dots,F_k$ 即可。

综上，我们可以将 $F_{v+1}$ 的问题用 $\mathcal{O}(k^2)$ 的复杂度转化为 $F_{\lfloor v/2\rfloor+1}$ 的问题，也就可以递归解决，最终 $\mathcal{O}(\log M)$ 层，求单个 $F_{x+1}$ 的复杂度为 $\mathcal{O}(k^2\log M)$ ，而我们要求 $k$ 个，所以乘以 $n$ 。复杂度是合理的。

综上，完成了 $F_{x+1}$ 的求解。

时间复杂度为 $\mathcal{O}(k^3\log M)$ 。中间求 $F_{x+1}$ 系数的部分如果使用 FFT 优化卷积的过程，可以进一步优化到 $\mathcal{O}(n^k)$ ，因为后面还是得高斯消元。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    k = II()
    v1 = LII()
    v2 = LII()
    
    mod = 10 ** 9 + 7
    
    xn = [[0] * k for _ in range(2 * k)]
    
    for i in range(k):
        xn[i][i] = 1
    
    for i in range(k, 2 * k):
        for j in range(k):
            for x in range(k):
                xn[i][x] += xn[i - j - 1][x] * v1[j]
                xn[i][x] %= mod
    
    def solve(x):
        if x < 2 * k: return xn[x]
        
        v = solve(x // 2)
        tag = x % 2
        
        res = [0] * (2 * k)
        
        for i in range(k):
            for j in range(k):
                res[i + j + tag] += v[i] * v[j]
                res[i + j + tag] %= mod
        
        ans = [0] * k
        
        for i in range(2 * k):
            for j in range(k):
                ans[j] += res[i] * xn[i][j]
                ans[j] %= mod
        
        return ans
    
    equations = [[0] * (k + 1) for _ in range(k)]
    
    for i in range(k):
        tmp = solve(v2[k - 1] - v2[i])
        for j in range(k):
            equations[j][i] = tmp[j]
    
    tmp = solve(v2[k - 1])
    for j in range(k):
        equations[j][k] = tmp[j]
    
    pt = 0
    
    for i in range(k):
        chosen = -1
        while pt < k + 1 and chosen == -1:
            for j in range(i, k):
                if equations[j][pt]:
                    chosen = j
                    break
            pt += 1
        
        pt -= 1
        
        if chosen != i:
            equations[i], equations[chosen] = equations[chosen], equations[i]
        
        if pt == k + 1: break
        
        v = pow(equations[i][pt], -1, mod)
        
        for j in range(pt, k + 1):
            equations[i][j] = equations[i][j] * v % mod
        
        for j in range(i + 1, k):
            w = (-equations[j][pt]) % mod
            for x in range(pt, k + 1):
                equations[j][x] += equations[i][x] * w
                equations[j][x] %= mod
    
    for i in range(k - 1, -1, -1):
        for j in range(k + 1):
            if equations[i][j]:
                
                for x in range(i):
                    w = (-equations[x][j]) % mod
                    for y in range(j, k + 1):
                        equations[x][y] += equations[i][y] * w
                        equations[x][y] %= mod
                
                break
    
    print(' '.join(str(equations[i][k]) for i in range(k)))
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int k, mod = 1e9 + 7;
	cin >> k;

	vector<int> v1(k), v2(k);
	for (auto &x: v1) cin >> x;
	for (auto &x: v2) cin >> x;

	vector<vector<int>> xn(2 * k, vector<int>(k, 0));

	for (int i = 0; i < k; i ++) xn[i][i] = 1;

	for (int i = k; i < 2 * k; i ++) {
		for (int j = 0; j < k; j ++) {
			for (int x = 0; x < k; x ++) {
				xn[i][x] = (xn[i][x] + 1ll * xn[i - j - 1][x] * v1[j]) % mod;
			}
		}
	}

	auto solve = [&] (auto &self, int x) -> vector<int> {
		if (x < 2 * k) return xn[x];

		auto v = self(self, x / 2);
		int tag = x % 2;

		vector<int> res(2 * k, 0);

		for (int i = 0; i < k; i ++) {
			for (int j = 0; j < k; j ++) {
				res[i + j + tag] = (res[i + j + tag] + 1ll * v[i] * v[j]) % mod;
			}
		}

		vector<int> ans(k, 0);

		for (int i = 0; i < 2 * k; i ++) {
			for (int j = 0; j < k; j ++) {
				ans[j] = (ans[j] + 1ll * res[i] * xn[i][j]) % mod;
			}
		}

		return ans;
	};

	vector<vector<int>> equations(k, vector<int>(k + 1));

	for (int i = 0; i < k; i ++) {
		auto tmp = solve(solve, v2[k - 1] - v2[i]);
		for (int j = 0; j < k; j ++) {
			equations[j][i] = tmp[j];
		}
	}

	{
		auto tmp = solve(solve, v2[k - 1]);
		for (int j = 0; j < k; j ++) {
			equations[j][k] = tmp[j];
		}
	}

	int pt = 0;

	for (int i = 0; i < k; i ++) {
		int chosen = -1;

		while (pt < k + 1 && chosen == -1) {
			for (int j = i; j < k; j ++) {
				if (equations[i][pt]) {
					chosen = j;
					break;
				}
			}
			pt ++;
		}

		pt --;

		if (chosen != i) swap(equations[i], equations[chosen]);
		if (pt == k + 1) break;

		int v = quickPow(equations[i][pt], mod - 2, mod);

		for (int j = pt; j <= k; j ++) {
			equations[i][j] = 1ll * equations[i][j] * v % mod;
		}

		for (int j = i + 1; j < k; j ++) {
			int w = (mod - equations[j][pt]) % mod;
			for (int x = pt; x <= k; x ++) {
				equations[j][x] = (equations[j][x] + 1ll * equations[i][x] * w) % mod;
			}
		}
	}

	for (int i = k - 1; i >= 0; i --) {
		for (int j = 0; j <= k; j ++) {
			if (equations[i][j]) {
				for (int x = 0; x < i; x ++) {
					int w = (mod - equations[x][j]) % mod;
					for (int y = j; y <= k; y ++) {
						equations[x][y] = (equations[x][y] + 1ll * equations[i][y] * w) % mod;
					}
				}
				break;
			}
		}
	}

	for (int i = 0; i < k; i ++) {
		cout << equations[i][k] << " \n"[i == k - 1];
	}

	return 0;
}
```
