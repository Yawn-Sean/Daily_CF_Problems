**提示 1：** 计算初始情况的角的个数。新增一个 $1$ 如何影响结果？

首先，我们可以计算初始情况的角的个数。为此，我们枚举中间那个点的位置，接下来分别统计同行同列有多少个其它的 $1$ ，两者数量相乘就是这个点对应的答案，全部相加即可。

新增一个 $1$ 呢？

- 其本身作为中心的点：此时答案新增为这一列当前 $1$ 的个数乘以这一行当前 $1$ 的个数。

- 中心的点与其同行：此时枚举这一行其它的 $1$ ，则这个 $1$ 对应的新增答案是中心 $1$ 的列中除了它的 $1$ 的个数。这些进行求和即可。因为整体静态，且修改为 $1$ 的位置原先必然不是 $1$ ，所以可以预处理。

- 中心的点与其同行：与前一种情况类似。

时间复杂度为 $\mathcal{O}(nm)$ 。

### 具体代码如下——

Python 做法如下——

```Python []
def main():
    t = II()
    outs = []
    
    rnd = random.getrandbits(30)
    
    for _ in range(t):
        n = II()
        v1 = LII()
        v2 = LII()
        
        ans = 0
        
        cnt = Counter()
        
        for i in range(n):
            ans += cnt[(v1[i] - v2[i]) ^ rnd]
            cnt[(v1[i] - v2[i]) ^ rnd] += 1
        
        cnt = Counter()
        
        for i in range(n):
            ans += cnt[(v1[i] + v2[i]) ^ rnd]
            cnt[(v1[i] + v2[i]) ^ rnd] += 1
        
        cnt = Counter()
        
        for i in range(n):
            ans -= cnt[(v1[i], v2[i])]
            cnt[(v1[i], v2[i])] += 1
        
        outs.append(ans)
    
    print('\n'.join(map(str, outs)))
```

C++ 做法如下——

```cpp []
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);

	int t;
	cin >> t;

	while (t --) {
		int n;
		cin >> n;

		vector<int> v1(n), v2(n);
		for (auto &x: v1) cin >> x;
		for (auto &x: v2) cin >> x;

		long long ans = 0;

		{
			map<int, int> cnt;
			for (int i = 0; i < n; i ++) {
				ans += cnt[v1[i] - v2[i]];
				cnt[v1[i] - v2[i]] ++;
			}
		}

		{
			map<int, int> cnt;
			for (int i = 0; i < n; i ++) {
				ans += cnt[v1[i] + v2[i]];
				cnt[v1[i] + v2[i]] ++;
			}
		}

		{
			map<pair<int, int>, int> cnt;
			for (int i = 0; i < n; i ++) {
				ans -= cnt[{v1[i], v2[i]}];
				cnt[{v1[i], v2[i]}] ++;
			}
		}

		cout << ans << '\n';
	}

	return 0;
}
```
